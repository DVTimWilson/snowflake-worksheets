    RESOURCE_HK as id

right_table as (

  select
    RESOURCE_HK as id

  from TIM_WILSON.SILVER.hub_resource

  where RESOURCE_HK is not null
    and 1=1

),

exceptions as (

  select
    left_table.id as left_id,
    right_table.id as right_id

  from left_table

  left join right_table
         on left_table.id = right_table.id

  where right_table.id is null

)

select * from exceptions

;
