WITH inventory_with_gaps AS (
    SELECT 
        UNIQUE_KEY
        , DATE
        , INVENTORY_ITEM_ID
        , INVENTORY_ITEM_DESCRIPTION
        , CLASS_ID
        , PROD_CODE
        , COREVIEW_PRODUCT_CATEGORY
        , PRODUCT_STAGE
        , PRODUCT_TYPE
        , ENTITY
        , BU
        , SITE
        , WAREHOUSE_CODE
        , EXTENDED_COST
        , UNIT_COST
        , UOM
        , QUANTITY
        , UNIT_WEIGHT
        , WEIGHT_LBS
        , IS_ACQUIRED_DATE
        , IS_ACQUIRED_NOW
        , META_LOAD_DATETIME
        , RECORD_SOURCE
        , MIN(DATE) OVER () AS MIN_DATE_OVERALL
        , MAX(DATE) OVER () AS MAX_DATE_OVERALL
        , LEAD(DATE, 1, CURRENT_DATE()) OVER (PARTITION BY INVENTORY_ITEM_ID, INVENTORY_ITEM_DESCRIPTION, ENTITY, SITE, WAREHOUSE_CODE ORDER BY DATE) AS LEAD_DATE
    FROM `joe_barter`.`gold_coreview`.`mart_inventory_stock_levels`
    WHERE INVENTORY_ITEM_ID NOT ILIKE 'SAMP%'
    -- AND ENTITY = 'ECS' AND UPPER(SITE) IN ('MEXIA', 'GREENVILLE')
    -- AND DATE BETWEEN '2026-07-10' AND '2026-07-13'
),

cross_join_data AS (
    SELECT cal.date_day AS dim_date, inv.*
    FROM `tim_wilson`.`silver`.`dim_calendar_usa` AS cal

    INNER JOIN inventory_with_gaps AS inv
    ON cal.date_day >= inv.DATE
    AND cal.date_day < inv.LEAD_DATE

    WHERE cal.date_day BETWEEN inv.MIN_DATE_OVERALL AND inv.MAX_DATE_OVERALL
)

SELECT
    dim_date
    , UNIQUE_KEY
    , DATE
    , INVENTORY_ITEM_ID
    , INVENTORY_ITEM_DESCRIPTION
    , CLASS_ID
    , PROD_CODE
    , COREVIEW_PRODUCT_CATEGORY
    , PRODUCT_STAGE
    , PRODUCT_TYPE
    , ENTITY
    , BU
    , SITE
    , WAREHOUSE_CODE
    , EXTENDED_COST
    , UNIT_COST
    , UOM
    , QUANTITY
    , UNIT_WEIGHT
    , WEIGHT_LBS
    , IS_ACQUIRED_DATE
    , IS_ACQUIRED_NOW
    , META_LOAD_DATETIME
    , RECORD_SOURCE
FROM cross_join_data
WHERE 1=1
    AND ENTITY = 'ECS' AND UPPER(SITE) IN ('MEXIA', 'GREENVILLE')
    AND DATE BETWEEN '2026-07-03' AND '2026-07-06'
ORDER BY dim_date DESC, entity, site, inventory_item_id, inventory_item_description
;
