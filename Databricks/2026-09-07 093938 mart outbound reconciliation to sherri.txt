SELECT x.PARTNUM, m.PRODUCT_ID, m.UNIQUE_KEY, *
FROM (
  SELECT
    CONCAT(CONCAT_WS('-', COMPANY, INVOICENUM, INVOICELINE), '-') AS UNIQUE_KEY,
    CONCAT(CONCAT(CONCAT(CONCAT(COMPANY, '-'), CAST(INVOICENUM AS STRING)), '-'), CAST(INVOICELINE AS STRING), '-') AS JOIN_PREFIX,
    *
  FROM production.silver.sat_outbound_invoice_kinetic_latest
  WHERE INVOICEDATE >= '2026-01-01'
) AS x

INNER JOIN production.gold_coreview.mart_outbound_transactions_pre_remap AS m
  ON m.ENTITY = x.COMPANY
  AND m.INVOICE_NUMBER = CAST(x.INVOICENUM AS STRING)
  AND startswith(m.UNIQUE_KEY || '-', x.JOIN_PREFIX)
--   AND m.UNIQUE_KEY NOT LIKE '%-CS'

WHERE x.PARTNUM != m.PRODUCT_ID
;
