WITH annual_sales AS (
  SELECT YEAR(invoice_date) AS year, entity, bu, SUM(sales_dollar) AS annual_sales
  FROM tim_wilson.silver.br_combined_outbound_invoice_line
  WHERE invoice_date IS NOT NULL
  GROUP BY YEAR(invoice_date), entity, bu
),

annual_sales_by_product_category AS (
  SELECT YEAR(invoice_date) AS year, entity, bu, product_id, SUM(sales_dollar) AS product_category_annual_sales
  FROM tim_wilson.silver.br_combined_outbound_invoice_line
  WHERE invoice_date IS NOT NULL
  GROUP BY YEAR(invoice_date), entity, bu, product_id
)

SELECT apc.*
  , as.annual_sales
  , apc.product_category_annual_sales / NULLIF(as.annual_sales, 0) AS proportion
  , ROUND(apc.product_category_annual_sales / NULLIF(as.annual_sales, 0) * 100, 2) AS percentage
FROM annual_sales_by_product_category AS apc
INNER JOIN annual_sales AS as
ON apc.year = as.year
AND apc.entity = as.entity
AND apc.bu = as.bu
ORDER BY apc.year, apc.entity, apc.bu, apc.product_id
;

WITH proportions AS (

  , entity
  , bu

  , SUM(sales_dollar) AS product_category_annual_sales
  , SUM(SUM(sales_dollar)) OVER (PARTITION BY YEAR(invoice_date), entity, bu) AS annual_sales
  , SUM(sales_dollar) / NULLIF(SUM(SUM(sales_dollar)) OVER (PARTITION BY YEAR(invoice_date), entity, bu), 0) AS proportion
  FROM tim_wilson.silver.br_combined_outbound_invoice_line
  WHERE invoice_date IS NOT NULL
  AND entity = 'PRC'
  GROUP BY YEAR(invoice_date), entity, bu, site, coreview_product_category
  ORDER BY year, entity, bu, site, coreview_product_category
),

sales_budgets AS (
  SELECT
    company_code
    , CASE WHEN UPPER(business_unit_name) = 'CONSTRUCTION' THEN 'OS'
        ELSE business_unit_name
      END AS business_unit_name
    , business_unit_key
    , goal_name
    , total_goal
    , date_sk
    , spread
    , date
    , month
    , year
  FROM tim_wilson.bronze.raw_sales_budgets
),

sales_budgets_with_proportions AS (
SELECT DISTINCT
  rst.year AS budget_year
  , rst.month AS budget_month
  , rst.company_code
  , rst.business_unit_name
  , pr.site
  , pr.coreview_product_category || ' Budget' AS coreview_product_category
  , pr.product_category_annual_sales AS product_category_annual_sales
  , pr.annual_sales AS annual_sales
  , pr.proportion AS proportion
  , rst.total_goal
  , rst.total_goal * pr.proportion AS site_plus_product_category_goal
FROM sales_budgets AS rst

LEFT OUTER JOIN proportions AS pr
ON rst.year = pr.year + 1
AND rst.company_code = pr.entity
AND rst.business_unit_name = pr.bu
)

SELECT *
FROM sales_budgets_with_proportions
WHERE company_code = 'PRC'
AND budget_year = 2024
AND budget_month = 1
ORDER BY budget_year, budget_month, company_code, business_unit_name, site, coreview_product_category
;
