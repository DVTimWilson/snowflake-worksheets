-- raw_department_dayforce

-- Original query 155 records
WITH deps as (
    SELECT DISTINCT ALLOCATED_DEPARTMENT
        FROM `tim_wilson`.`BRONZE`.`src_allocatedhoursandearnings`
),

deps_details AS (
    SELECT DISTINCT deps.ALLOCATED_DEPARTMENT
                  , a.ALLOCATED_LEDGER_CODE
                  , a.DEPT
                  , a.ENTITY
                  , a.SITE
                  , a.EXTRACT_DATE
                  , ROW_NUMBER() OVER(PARTITION BY deps.ALLOCATED_DEPARTMENT ORDER BY DEPT ASC) AS rn
    FROM `tim_wilson`.`BRONZE`.`src_allocatedhoursandearnings` as a
    INNER JOIN deps
        ON deps.allocated_department = a.allocated_department
)

SELECT *
FROM deps_details
WHERE rn = 1
ORDER BY 1
;
