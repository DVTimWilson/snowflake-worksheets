WITH
mart_inventory_stock_levels AS (
  SELECT 
        UNIQUE_KEY
      , DATE
      , INVENTORY_ITEM_ID
      , INVENTORY_ITEM_DESCRIPTION
      , CLASS_ID
      , PROD_CODE
      , COREVIEW_PRODUCT_CATEGORY
      , ENTITY
      , BU
      , SITE
      , WAREHOUSE_CODE
      , EXTENDED_COST
      , UNIT_COST
      , UOM
      , QUANTITY
      , UNIT_WEIGHT
      , WEIGHT_LBS
      , IS_ACQUIRED_DATE
      , IS_ACQUIRED_NOW
      , META_LOAD_DATETIME
      , RECORD_SOURCE
  FROM uat.silver.bsat_combined_inventory_stock_levels
  WHERE DATE IS NOT NULL
),

mart_inventory_usage AS (
  SELECT 
      UNIQUE_KEY
    , COMPANY
    , TRANNUM
    , PARTNUM
    , WAREHOUSECODE
    , BINNUM
    , TRANCLASS
    , TRANTYPE
    , INVENTORYTRANS
    , TRANDATE
    , CASE WHEN TRANCLASS = 'I' THEN TRANQTY * -1 ELSE TRANQTY END AS TRANQTY
    , UM
    , JOBNUM
    , PACKLINE
    , PONUM
    , PACKNUM
    , POLINE
    , PORELNUM
    , ORDERNUM
    , ORDERLINE
    , ORDERRELNUM
    , BINNUM2
    , PARTDESCRIPTION
    , REVISIONNUM
    , VENDORNUM
    , INVOICENUM
    , INVOICELINE
    , FISCALYEAR
    , FISCALPERIOD
    , SYSDATE
    , SYSROWID
    , MTLUNITCOST
    , LBRUNITCOST
    , BURUNITCOST
    , SUBUNITCOST
    , MTLBURUNITCOST
    , EXTCOST
    , COSTMETHOD
    , INVADJSRC
    , INVADJREASON
    , PLANT
    , GLTRANS
    , POSTEDTOGL
    , RECORD_SOURCE
    , SITE
    , QUICKBOOKSITEMTYPE
    , QUICKBOOKSNAME
    , QUICKBOOKSMEMO
FROM uat.silver.bsat_combined_inventory_transactions
)

SELECT DATE
, ANY_VALUE(INVENTORY_ITEM_ID) AS INVENTORY_ITEM_ID
, ANY_VALUE(OPENING_STOCK) AS OPENING_STOCK
, ANY_VALUE(CLOSING_STOCK) AS CLOSING_STOCK
, ANY_VALUE(CLOSING_STOCK) - ANY_VALUE(OPENING_STOCK) AS STOCK_MOVEMENT
, SUM(TRANQTYR) AS RECEIPT
, SUM(TRANQTYI) AS ISSUE
, SUM(TRANQTYA) AS ADJUSTMENTS
, ANY_VALUE(OPENING_STOCK) + SUM(TRANQTYR) + SUM(TRANQTYI) + SUM(TRANQTYA) - ANY_VALUE(CLOSING_STOCK) AS ADJUSTMENTSBALANCING
FROM (
SELECT date_day
, sO.DATE AS DATE
, sO.INVENTORY_ITEM_ID
, sO.QUANTITY - LAG(sO.QUANTITY) OVER (PARTITION BY sO.INVENTORY_ITEM_ID ORDER BY sO.DATE) AS STOCK_MOVEMENT
, sO.QUANTITY AS OPENING_STOCK
, sC.QUANTITY AS CLOSING_STOCK
, COALESCE(tR.TRANQTY, 0) AS TRANQTYR
, COALESCE(tI.TRANQTY, 0) AS TRANQTYI
, COALESCE(tA.TRANQTY, 0) AS TRANQTYA
, LAG(sO.DATE) OVER (PARTITION BY sO.INVENTORY_ITEM_ID ORDER BY sO.DATE) AS PREV_DATE
, sO.SITE
FROM uat.silver.dim_calendar_usa AS c

-- Opening stock
LEFT OUTER JOIN mart_inventory_stock_levels AS sO
ON date_day - 1 = sO.DATE

-- Closing stock
LEFT OUTER JOIN mart_inventory_stock_levels AS sC
ON date_day = sC.DATE
AND sO.INVENTORY_ITEM_ID = sC.INVENTORY_ITEM_ID

-- Inventory transaction Receipts
LEFT OUTER JOIN (SELECT PARTNUM, TRANDATE, SITE, COMPANY, SUM(TRANQTY) AS TRANQTY FROM mart_inventory_usage WHERE TRANCLASS = 'R' GROUP BY PARTNUM, TRANDATE, SITE, COMPANY) AS tR
ON sO.INVENTORY_ITEM_ID = tR.PARTNUM
AND date_day - 1 = TO_DATE(tR.TRANDATE)
AND sO.SITE = tR.SITE
AND sO.ENTITY = tR.COMPANY

-- Inventory transaction Issues
LEFT OUTER JOIN (SELECT PARTNUM, TRANDATE, SITE, COMPANY, SUM(TRANQTY) AS TRANQTY FROM mart_inventory_usage WHERE TRANCLASS = 'I' GROUP BY PARTNUM, TRANDATE, SITE, COMPANY) AS tI
ON sO.INVENTORY_ITEM_ID = tI.PARTNUM
AND date_day - 1 = TO_DATE(tI.TRANDATE)
AND sO.SITE = tI.SITE
AND sO.ENTITY = tI.COMPANY

-- Inventory transaction Adjustments
LEFT OUTER JOIN (SELECT PARTNUM, TRANDATE, SITE, COMPANY, SUM(TRANQTY) AS TRANQTY FROM mart_inventory_usage WHERE TRANCLASS = 'A' AND TRANTYPE != 'MFG-VAR' GROUP BY PARTNUM, TRANDATE, SITE, COMPANY) AS tA
ON sO.INVENTORY_ITEM_ID = tA.PARTNUM
AND date_day - 1 = TO_DATE(tA.TRANDATE)
AND sO.SITE = tA.SITE
AND sO.ENTITY = tA.COMPANY

WHERE 1 = 1 
AND sO.INVENTORY_ITEM_ID = 'L000036-04-89-1903'
-- AND sC.INVENTORY_ITEM_ID = 'L000036-04-89-1903'
-- AND sO.INVENTORY_ITEM_ID = '1009015-06-01-0400'
-- AND sO.INVENTORY_ITEM_ID = '1002529-04-01-1006'
-- AND sO.INVENTORY_ITEM_ID IN (SELECT DISTINCT PARTNUM FROM mart_inventory_usage)
ORDER BY sO.INVENTORY_ITEM_ID, sO.DATE

) AS x
GROUP BY INVENTORY_ITEM_ID, x.DATE
ORDER BY INVENTORY_ITEM_ID, DATE
;
