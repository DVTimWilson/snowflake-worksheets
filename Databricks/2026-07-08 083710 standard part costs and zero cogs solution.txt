SELECT COUNT(*)
FROM (
SELECT 
     kin.INVOICE_LINE_LHK
     , kin.UNIQUE_KEY
     , kin.INVOICELINE
     , CAST(kin.INVOICENUM AS string) AS INVOICE_NUMBER
     , kin.INVOICEDATE AS INVOICE_DATE
     , kin.NAME AS CUSTOMER
     , kin.PARTNUM AS PRODUCT_ID
     , kin.LINEDESC AS PRODUCT_DESCRIPTION
     , COALESCE(NULLIF(cpm.L2_CATEGORY, ''), 'Uncategorized') AS COREVIEW_PRODUCT_CATEGORY
     , COALESCE(NULLIF(cpm.L3_STAGE, ''), 'Uncategorized') AS PRODUCT_STAGE
     , COALESCE(NULLIF(cpm.L4_TYPE, ''), 'Uncategorized') AS PRODUCT_TYPE  
     , kin.PLANT AS PLANT
     , COALESCE(NULLIF(plant.CITY, ''), 'Uncategorized') AS SITE
     , COALESCE(NULLIF(cpm.L1_BU, ''), 'Uncategorized') AS BU
     , 
CASE
  WHEN UPPER(TRIM(kin.COMPANY)) <> 'EII' AND mec.BUSINESS_SEGMENT = 'Materials Business Unit (MBU)' THEN 'MBU'
  WHEN UPPER(TRIM(kin.COMPANY)) <> 'EII' AND mec.BUSINESS_SEGMENT = 'Outdoor Surfacing'           THEN 'OSG'
  WHEN UPPER(TRIM(kin.BUSINESS_UNIT_ID)) = 'MTL'                                                                THEN 'MBU'
  WHEN UPPER(TRIM(kin.COMPANY)) =  'EII' AND (UPPER(TRIM(kin.BUSINESS_UNIT_ID)) IS NULL OR UPPER(TRIM(kin.BUSINESS_UNIT_ID)) <> 'F&I')         THEN 'MBU'
  WHEN UPPER(TRIM(kin.BUSINESS_UNIT_ID)) IS NULL OR UPPER(TRIM(kin.BUSINESS_UNIT_ID)) IN ('','-1')                                    THEN 'Uncategorized'
  ELSE UPPER(TRIM(kin.BUSINESS_UNIT_ID))
END AS BU_LOGICAL
     , kin.COMPANY AS ENTITY
     , kin.DOCEXTPRICE AS SALES_DOLLAR
     , kin.SALESPRICE AS SALES_DOLLAR_WITH_FREIGHT
     , CASE WHEN kin.DOCEXTCOST = 0 THEN stdcosts.STD_TOTAL_COST ELSE kin.DOCEXTCOST END  AS COGS_DOLLAR
     , kin.DOCUNITCOST AS UNIT_COST
     , kin.GM AS GROSS_MARGIN_DOLLAR
     , kin.GMPCT AS GROSS_MARGIN_PCT
     , kin.SELLINGSHIPQTY as QUANTITY
     , CASE
  WHEN UPPER(TRIM(pm.PART_IUM)) IS NULL OR UPPER(TRIM(pm.PART_IUM)) IN ('', '-1', '0', 'NULL') THEN 'Uncategorized'
  ELSE COALESCE(
         element_at(
           map_from_arrays(
             array('EACH', 'LOAD'),
             array('EA', 'LD')
           ),
           UPPER(TRIM(pm.PART_IUM))
         ),
         UPPER(TRIM(pm.PART_IUM))
       )
END AS UOM
     , 

(
  
    CAST(COALESCE(TRY_MULTIPLY(kin.SELLINGSHIPQTY, pm.PART_NETWEIGHT) , 0) AS DECIMAL(38,8))
  
) AS WEIGHT_LBS
     , 'NA' AS COGSPC_SOURCE
     , kin.LOAD_DATETIME
     , kin.RECORD_SOURCE
     , kin.HASHDIFF
FROM `tim_wilson`.`silver`.`sat_outbound_invoice_kinetic_latest` as kin

INNER JOIN (
    SELECT y.Business_Segment 
           , y.Company_Code
    FROM `tim_wilson`.`silver`.`ref_master_ecore_companies` AS y
) AS mec
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(mec.company_code)

LEFT OUTER JOIN (
  SELECT PART_PARTNUM
    , PART_IUM
    , PART_NETWEIGHT
    , PART_COMPANY 
  FROM `tim_wilson`.`silver`.`ref_sat_kinetic_part_master_latest`
) AS pm
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.PARTNUM) = UPPER(pm.PART_PARTNUM)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(pm.PART_COMPANY)

LEFT OUTER JOIN `tim_wilson`.`silver`.`sat_plant_kinetic_latest` as plant
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(plant.COMPANY)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.PLANT) = UPPER(plant.PLANT)

LEFT OUTER JOIN `tim_wilson`.`silver`.`ref_sat_consolidated_part_master_latest` AS cpm
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.PARTNUM) = UPPER(cpm.PART_NUM)
        AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(cpm.COMPANY)
        AND -- Case-insensitive equality for ad-hoc use
UPPER(cpm.SYSTEM) = UPPER('KINETIC')

LEFT OUTER JOIN `tim_wilson`.`silver`.`ref_sat_bu_site_part_costs_latest` as stdcosts
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(stdcosts.COMPANY)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(COALESCE(NULLIF(plant.CITY, ''), 'Uncategorized')) = UPPER(stdcosts.SITE)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.PARTNUM) = UPPER(stdcosts.PART)
) AS x
;
