, SUM(PARTBIN_ONHANDQTY) AS PARTBIN_ONHANDQTY
, ANY_VALUE(LOAD_DATETIME) AS LOAD_DATETIME
, ANY_VALUE(RECORD_SOURCE) AS RECORD_SOURCE

FROM `tim_wilson`.`silver`.`sat_inventory_stock_levels_latest`

-- WHERE PARTBIN_PARTNUM = 'SBA-339835'
WHERE PARTBIN_PARTNUM = '1008587-04-90-1521'
AND INVENTORY_DATE = '2026-02-28'

GROUP BY INVENTORY_DATE
, PARTBIN_COMPANY
, PLANT_PLANT
, UPPER(PARTWHSE_WAREHOUSECODE)
, PARTBIN_PARTNUM

ORDER BY INVENTORY_DATE, PARTBIN_COMPANY, PLANT_PLANT, PARTWHSE_WAREHOUSECODE

WITH feb AS (
SELECT ENTITY, CAST(SUM(EXTENDED_COST) AS DECIMAL(18,2)) AS VALUE, CAST(SUM(WEIGHT_LBS) AS DECIMAL(18,2)) AS WEIGHT, COUNT(*) AS RECORD_COUNT
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels
WHERE DATE = '2026-02-28'
GROUP BY ROLLUP (ENTITY)
ORDER BY ENTITY
),

mar AS (
SELECT ENTITY, CAST(SUM(EXTENDED_COST) AS DECIMAL(18,2)) AS VALUE, CAST(SUM(WEIGHT_LBS) AS DECIMAL(18,2)) AS WEIGHT, COUNT(*) AS RECORD_COUNT
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels
WHERE DATE = '2026-03-26'
GROUP BY ROLLUP (ENTITY)
ORDER BY ENTITY
)

SELECT a.entity
, b.VALUE - a.VALUE AS VALUE_DIFF
, b.WEIGHT - a.WEIGHT AS WEIGHT_DIFF
FROM feb AS a
CROSS JOIN mar AS b
WHERE a.ENTITY = b.ENTITY
ORDER BY a.ENTITY
;
