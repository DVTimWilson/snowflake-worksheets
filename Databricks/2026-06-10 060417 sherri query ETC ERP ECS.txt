SELECT *
FROM landing.kinetic.sales_sherrigmquery_allcompanies
WHERE Company IN ('ETC', 'ERP', 'ECS')

        QUALIFY
                RANK() OVER (PARTITION BY INVOICENUM
                                        ,INVOICETYPE
                                        ,INVOICELINE
                                        ,COMPANY
                                        ,INVOICEDATE 
                                   ORDER BY EXTRACT_DATE DESC) = 1
            AND
                ROW_NUMBER() OVER (PARTITION BY Company
                                        , InvoiceNum
                                        , InvoiceDate
                                        , InvoiceType
                                        , Name
                                        , InvoiceLine
                                        , SellingShipQty
                                        , DocUnitPrice
                                        , DocTotalMiscChrg
                                        , DocUnitCost
                                        , ShipToCustNum
                                        , ShipToAddress1
                                        , ShipToAddress2
                                        , SoldToState
                                        , GroupCode
                                        , TerritoryID
                                        , Character03
                                        , Formula
                                        , Width
                                        , SFSold
                                        , Market2_c
                                        , ProjectName_c
                                        , BillToCountry
                                        , RequestDate
                                        , NeedByDate
                                        , CustSegment_c 
                                   ORDER BY EXTRACT_DATE DESC) = 1

ORDER BY Company, InvoiceDate, InvoiceNum, InvoiceLine
;
