FROM (
SELECT date_day
, s.DATE
, s.INVENTORY_ITEM_ID
, s.QUANTITY
, s.QUANTITY - LAG(s.QUANTITY) OVER (PARTITION BY INVENTORY_ITEM_ID ORDER BY DATE) AS STOCK_MOVEMENT
, COALESCE(t.TRANQTY, 0) AS TRANQTY
, LAG(s.DATE) OVER (PARTITION BY INVENTORY_ITEM_ID ORDER BY DATE) AS PREV_DATE
FROM uat.silver.dim_calendar_usa

LEFT OUTER JOIN mart_inventory_stock_levels AS s
ON date_day = s.DATE

LEFT OUTER JOIN mart_inventory_usage AS t
ON s.INVENTORY_ITEM_ID = t.PARTNUM
-- AND s.DATE = TO_DATE(t.TRANDATE)
AND date_day = TO_DATE(t.TRANDATE)
AND s.SITE = t.SITE
AND S.ENTITY = t.COMPANY

WHERE s.INVENTORY_ITEM_ID = 'L000036-04-89-1903'
-- WHERE s.INVENTORY_ITEM_ID = '1009015-06-01-0400'
-- WHERE s.INVENTORY_ITEM_ID = '1002529-04-01-1006'
-- WHERE s.INVENTORY_ITEM_ID IN (SELECT DISTINCT PARTNUM FROM mart_inventory_usage)
ORDER BY s.INVENTORY_ITEM_ID, s.DATE

) AS x
GROUP BY INVENTORY_ITEM_ID, DATE
HAVING NOT (STOCK_MOVEMENT = 0 AND TRANSACTION_QUANTITY = 0)
ORDER BY INVENTORY_ITEM_ID, DATE
;
