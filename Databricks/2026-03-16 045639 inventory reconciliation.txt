SELECT *
FROM landing.kinetic.stockstat
WHERE PartBin_PartNum = 'L000036-04-89-1903'
AND Extract_Date BETWEEN '2025-10-01' AND '2026-01-31'
ORDER BY Extract_Date, PartBin_BinNum
;

SELECT *
FROM landing.kinetic.parttran
WHERE PartNum = 'L000036-04-89-1903'
AND TranDate BETWEEN '2025-10-01' AND '2026-01-31'
ORDER BY TranDate, TranNum
;
