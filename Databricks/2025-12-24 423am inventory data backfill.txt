CREATE OR REPLACE TABLE tim_wilson.meta.DIM_CALENDAR_US AS
(

WITH RECURSIVE cte_dates (dim_date) MAX RECURSION LEVEL 12000 AS (
  SELECT TO_DATE('2020-01-01')
  UNION ALL
  SELECT TO_DATE(DATEADD(day, 1, dim_date))
  FROM  cte_dates
  WHERE dim_date < TO_DATE('2050-12-31')
),

years AS (
    SELECT DATE_PART('year', dim_date) AS cal_year
    FROM cte_dates
    GROUP BY cal_year
),

federal_holidays AS (
SELECT
        MAKE_DATE(cal_year, 1, 1) AS new_years_day
        ,DATEADD(
			day,
			MOD( 7 + 1 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 1, 1)) ), 7) + 14,
			MAKE_DATE(cal_year, 1, 1)
			) AS birthday_of_martin_luther_king_jr
        ,DATEADD(
			day,
			MOD( 7 + 1 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 2, 1)) ), 7) + 14,
			MAKE_DATE(cal_year, 2, 1)
			) AS washingtons_birthday
        ,DATEADD(
			day,
			MOD( 7 + 1 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 6, 1)) ), 7) - 7,
			MAKE_DATE(cal_year, 6, 1)
			) AS memorial_day
        ,CASE WHEN cal_year >= 2021 THEN
            MAKE_DATE(cal_year, 6, 19)
			ELSE NULL
         END AS juneteenth
        ,MAKE_DATE(cal_year, 7, 4) AS independence_day
        ,DATEADD(
			day,
			MOD( 7 + 1 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 9, 1)) ), 7),
			MAKE_DATE(cal_year, 9, 1)
			) AS labor_day
        ,DATEADD(
			day,
			MOD( 7 + 1 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 10, 1)) ), 7) + 7,
			MAKE_DATE(cal_year, 10, 1)
			) AS columbus_day
        ,MAKE_DATE(cal_year, 11, 11) AS veterans_day
        ,DATEADD(
			day,
			MOD( 7 + 4 - DATE_PART('dayofweek_iso', DATE_TRUNC('month', MAKE_DATE(cal_year, 11, 1)) ), 7) + 21,
			MAKE_DATE(cal_year, 11, 1)
			) AS thanksgiving_day
        ,MAKE_DATE(cal_year, 12, 25) AS christmas_day
    FROM years
),

federal_holidays_observed AS (
    SELECT
        CASE WHEN DAYNAME(new_years_day) = 'Sat' THEN DATEADD(day, -1, new_years_day) WHEN DAYNAME(new_years_day) = 'Sun' THEN DATEADD(day, 1, new_years_day) ELSE new_years_day END AS new_years_day
        ,DAYNAME(CASE WHEN DAYNAME(new_years_day) = 'Sat' THEN DATEADD(day, -1, new_years_day) WHEN DAYNAME(new_years_day) = 'Sun' THEN DATEADD(day, 1, new_years_day) ELSE new_years_day END) AS new_years_day_test
        ,birthday_of_martin_luther_king_jr
        ,washingtons_birthday
        ,memorial_day
        ,CASE WHEN DAYNAME(juneteenth) = 'Sat' THEN DATEADD(day, -1, juneteenth) WHEN DAYNAME(juneteenth) = 'Sun' THEN DATEADD(day, 1, juneteenth) ELSE juneteenth END AS juneteenth
        ,DAYNAME(CASE WHEN DAYNAME(juneteenth) = 'Sat' THEN DATEADD(day, -1, juneteenth) WHEN DAYNAME(juneteenth) = 'Sun' THEN DATEADD(day, 1, juneteenth) ELSE juneteenth END) AS juneteenth_test
        ,CASE WHEN DAYNAME(independence_day) = 'Sat' THEN DATEADD(day, -1, independence_day) WHEN DAYNAME(independence_day) = 'Sun' THEN DATEADD(day, 1, independence_day) ELSE independence_day END AS independence_day
        ,DAYNAME(CASE WHEN DAYNAME(independence_day) = 'Sat' THEN DATEADD(day, -1, independence_day) WHEN DAYNAME(independence_day) = 'Sun' THEN DATEADD(day, 1, independence_day) ELSE independence_day END) AS independence_day_test
        ,labor_day
        ,columbus_day
        ,CASE WHEN DAYNAME(veterans_day) = 'Sat' THEN DATEADD(day, -1, veterans_day) WHEN DAYNAME(veterans_day) = 'Sun' THEN DATEADD(day, 1, veterans_day) ELSE veterans_day END AS veterans_day
        ,DAYNAME(CASE WHEN DAYNAME(veterans_day) = 'Sat' THEN DATEADD(day, -1, veterans_day) WHEN DAYNAME(veterans_day) = 'Sun' THEN DATEADD(day, 1, veterans_day) ELSE veterans_day END) AS veterans_day_test
        ,thanksgiving_day
        ,CASE WHEN DAYNAME(christmas_day) = 'Sat' THEN DATEADD(day, -1, christmas_day) WHEN DAYNAME(christmas_day) = 'Sun' THEN DATEADD(day, 1, christmas_day) ELSE christmas_day END AS christmas_day
        ,DAYNAME(CASE WHEN DAYNAME(christmas_day) = 'Sat' THEN DATEADD(day, -1, christmas_day) WHEN DAYNAME(christmas_day) = 'Sun' THEN DATEADD(day, 1, christmas_day) ELSE christmas_day END) AS christmas_day_test
    FROM federal_holidays
),

federal_holidays_unpivot AS (
    SELECT holiday_date, holiday_name
    FROM
    (
    SELECT
        new_years_day
        ,birthday_of_martin_luther_king_jr
        ,washingtons_birthday
        ,memorial_day
        ,juneteenth
        ,independence_day
        ,labor_day
        ,columbus_day
        ,veterans_day
        ,thanksgiving_day
        ,christmas_day
    FROM federal_holidays_observed
    ) AS unpivot_source
    UNPIVOT
    (
        holiday_date FOR holiday_name IN
            (
                new_years_day
                ,birthday_of_martin_luther_king_jr
                ,washingtons_birthday
                ,memorial_day
                ,juneteenth
                ,independence_day
                ,labor_day
                ,columbus_day
                ,veterans_day
                ,thanksgiving_day
                ,christmas_day
            )
    ) AS unpivot
),

calendar_dim AS (
    SELECT
        dim_date
        ,DATE_TRUNC('week', dim_date) AS week_commencing_date
        ,DATE_FORMAT(dim_date, 'yyyyMM') AS year_month
        ,CASE WHEN DAYNAME(dim_date) IN ('Sat', 'Sun') THEN 0
            WHEN holiday_date IS NOT NULL THEN 0
            ELSE 1
        END AS working_day_flag
        ,SUM(CASE WHEN DAYNAME(dim_date) IN ('Sat', 'Sun') THEN 0
            WHEN holiday_date IS NOT NULL THEN 0
            ELSE 1
        END) OVER(ORDER BY dim_date) AS working_day_index
        ,holiday_name
        ,DECODE(DAYNAME(dim_date),
                     'Mon', 'Monday',
                     'Tue', 'Tuesday',
                     'Wed', 'Wednesday',
                     'Thu', 'Thursday',
                     'Fri', 'Friday',
                     'Sat', 'Saturday',
                     'Sun', 'Sunday')  AS day_name
        ,DAYNAME(dim_date) AS day_name_abv
        ,DAYOFWEEK(dim_date) AS day_of_week
        ,MONTH(dim_date) AS month_num
        ,CONCAT(YEAR(dim_date),TO_CHAR(dim_date,'MM')) AS month_id
        ,TO_CHAR(dim_date,'MMMM') AS month_name
        ,MONTHNAME(dim_date) AS month_name_abv
        ,TO_DATE(DATE_TRUNC('MONTH', dim_date)) as first_day_month
        ,LAST_DAY(dim_date) AS last_day_month
        ,QUARTER(dim_date) AS quarter_num
        ,CONCAT(YEAR(dim_date),'Q',QUARTER(dim_date)) AS quarter_id
        ,TO_DATE(DATE_TRUNC('QUARTER', dim_date)) as first_day_quarter
        ,TO_DATE(DATE_ADD(QUARTER, 1, DATE_ADD(DAY, -1, TO_DATE(DATE_TRUNC('QUARTER', dim_date))))) as last_day_quarter
        ,YEAR(dim_date) AS dim_year
    FROM cte_dates
    LEFT OUTER JOIN federal_holidays_unpivot
    ON dim_date = holiday_date
)

SELECT *
FROM calendar_dim
ORDER BY dim_date ASC
)
;

WITH inventory_with_gaps AS (
    SELECT ItemType
        , Date
        , Item
        , OnHand
        , UOM
        , AvgCost
        , AssetValue
        , Export_date_from_QB
        , Extract_Date
    FROM landing.quickbooks.stock_status_360
    QUALIFY ROW_NUMBER() OVER(PARTITION BY ItemType, Item, Date ORDER BY Extract_Date DESC) = 1
),

driver AS (
    SELECT calendar.dim_date
        , inventory.itemtype AS dim_itemtype
        , inventory.item AS dim_item
    FROM tim_wilson.meta.DIM_CALENDAR_US AS calendar

    CROSS JOIN inventory_with_gaps AS inventory
),

rows_to_process AS (
    SELECT DISTINCT driver.dim_date, driver.dim_itemtype, driver.dim_item, inventory.*
    FROM driver

    LEFT OUTER JOIN inventory_with_gaps AS inventory
    ON driver.dim_Date = inventory.date
    AND driver.dim_itemtype = inventory.itemtype
    AND driver.dim_item = inventory.item    
)

SELECT dim_date
    , dim_itemtype
    , dim_item
    , Date
    , Itemtype
    , Item
    , OnHand
    , CASE WHEN OnHand IS NULL THEN LAG(OnHand) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item ORDER BY dim_date)
      ELSE NULL
      END AS BackfillOnHand
    , UOM
    , AvgCost
    , AssetValue
    , Export_date_from_QB
    , Extract_Date
FROM rows_to_process
WHERE 1=1
AND dim_item = '01 Inventory Parts (Inventory Part)' 
AND dim_date >= '2024-10-08'
ORDER BY dim_date ASC, dim_itemtype, dim_item
;
