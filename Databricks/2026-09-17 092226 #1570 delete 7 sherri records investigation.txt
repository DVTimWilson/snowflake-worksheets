    SELECT INVOICE_LINE_LHK
    FROM production.silver.sat_outbound_invoice_kinetic

    760422
    , 765351
    , 767268
    , 767291
    , 779351
    , 787085
    )
    OR (
    InvoiceNum = 767559
AND InvoiceLine = 1
    )
)
;
