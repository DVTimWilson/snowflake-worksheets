SELECT x.COMPANY, COUNT(*), ANY_VALUE(z.zcount)
FROM (

SELECT t.COMPANY, t.TRANNUM, t.PARTNUM, t.TRANQTY, t.EXTCOST
, i.PRODUCT_ID, i.QUANTITY, i.UNIT_COST
FROM tim_wilson.gold_coreview.mart_inventory_transactions AS t

LEFT OUTER JOIN tim_wilson.gold_coreview.mart_outbound_transactions AS i
ON t.TRANNUM = i.INVOICE_NUMBER

WHERE 1 = 1 
AND t.RECORD_SOURCE ILIKE 'QUICK%'
AND t.TRANTYPE = 'Invoice'
AND i.QUANTITY != 0
AND TRY_DIVIDE(ABS(t.EXTCOST - i.UNIT_COST), t.EXTCOST) > 0.05
ORDER BY TRY_DIVIDE(ABS(t.EXTCOST - i.UNIT_COST), t.EXTCOST) DESC, t.TRANDATE DESC, t.TRANNUM ASC
) AS x

INNER JOIN (SELECT COMPANY, COUNT(*) AS zcount FROM tim_wilson.gold_coreview.mart_inventory_transactions GROUP BY COMPANY) AS z
ON x.company = z.company

GROUP BY x.COMPANY
;
