-- Partitions with duplicate rows on the most recent DATE where QUANTITY values
-- are mixed (some > 0, some <= 0). These cause non-deterministic ROW_NUMBER()
-- results when ORDER BY DATE DESC alone is used.
WITH ranked AS (
  SELECT
    ENTITY,
    INVENTORY_ITEM_ID,
    DATE,
    QUANTITY,
    DENSE_RANK() OVER (PARTITION BY ENTITY, INVENTORY_ITEM_ID ORDER BY DATE DESC) AS dr
  FROM develop.gold_coreview.mart_inventory_stock_levels
),
top_date_ties AS (
  SELECT
    ENTITY,
    INVENTORY_ITEM_ID,
    DATE AS latest_date,
    COUNT(*) AS duplicate_rows,
    SUM(CASE WHEN QUANTITY > 0 THEN 1 ELSE 0 END) AS rows_qty_positive,
    SUM(CASE WHEN QUANTITY <= 0 OR QUANTITY IS NULL THEN 1 ELSE 0 END) AS rows_qty_zero_or_neg,
    MIN(QUANTITY) AS min_quantity,
    MAX(QUANTITY) AS max_quantity
  FROM ranked
  WHERE dr = 1
  GROUP BY ENTITY, INVENTORY_ITEM_ID, DATE
  HAVING COUNT(*) > 1
    AND SUM(CASE WHEN QUANTITY > 0 THEN 1 ELSE 0 END) > 0
    AND SUM(CASE WHEN QUANTITY <= 0 OR QUANTITY IS NULL THEN 1 ELSE 0 END) > 0
)
SELECT *
FROM top_date_ties
ORDER BY duplicate_rows DESC, ENTITY, INVENTORY_ITEM_ID
;
