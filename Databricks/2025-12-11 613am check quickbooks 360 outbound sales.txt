,ROW_NUMBER() OVER (
        PARTITION BY 
              UPPER(COMPANY)
            , UPPER(NUM)
            , UPPER(NAME)
            , UPPER(ITEM)
            , UPPER(MEMO)
            , QTY
            , INDEX
        ORDER BY EXTRACT_DATE DESC)

WHERE DATE = '2025-11-30'
AND NUM = 'MLN'
ORDER BY AMOUNT, Export_Date_From_QB
