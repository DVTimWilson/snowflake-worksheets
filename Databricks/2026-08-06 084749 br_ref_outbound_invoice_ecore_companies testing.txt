    3, '%TSS%',                           'TSS',   DATE'1900-01-01', DATE'9999-12-31'
  ) AS (priority, pattern, code, valid_from, valid_to)
),

scored AS (
  SELECT
    d.INVOICE_COMPANY_NAME
    , p.code AS INTERNAL_COMPANY_CODE
    , x.ACQUISITION_DATE
    , ROW_NUMBER() OVER (
        PARTITION BY d.INVOICE_COMPANY_NAME, d.INVOICE_DATE
        ORDER BY p.priority
      ) AS rn
  FROM invoice_data AS d
  INNER JOIN all_patterns AS p
    ON d.INVOICE_COMPANY_NAME LIKE p.pattern
    AND d.INVOICE_DATE >= p.valid_from
    AND d.INVOICE_DATE <= p.valid_to
  LEFT OUTER JOIN landing.reftables.MASTER_ECORE_COMPANIES AS x
    ON UPPER(p.code) = UPPER(x.COMPANY_CODE)
)

SELECT DISTINCT
  INVOICE_COMPANY_NAME
  , INTERNAL_COMPANY_CODE
  , ACQUISITION_DATE
FROM scored
WHERE rn = 1
  AND INVOICE_COMPANY_NAME NOT LIKE '%DO NOT USE%'
ORDER BY INVOICE_COMPANY_NAME
;
