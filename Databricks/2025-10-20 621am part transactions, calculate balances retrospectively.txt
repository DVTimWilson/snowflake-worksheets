SELECT -- MIN(PartTran_TranDate), MAX(PartTran_TranDate)
PartTran_TranDate
, DATE(PartTran_TranDate)
, PartTran_PartNum
, PartTran_TranQty * -1
, SUM(PartTran_TranQty * -1) OVER (PARTITION BY PartTran_PartNum ORDER BY DATE(PartTran_TranDate) ASC RANGE BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS total_movement_to_date

WHERE 1 = 1
AND UPPER(PartTran_PartNum) = 'RMF-000003'
AND DATE(PartTran_TranDate) BETWEEN '2024-01-02' AND '2024-01-08'
-- AND DATE(PartTran_TranDate) = '2024-01-02'
