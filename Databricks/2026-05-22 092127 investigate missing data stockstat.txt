SELECT DATE, COUNT(*)
FROM (
SELECT 
      UNIQUE_KEY
    , DATE
    , INVENTORY_ITEM_ID
    , INVENTORY_ITEM_DESCRIPTION
    , CLASS_ID
    , PROD_CODE
    , COREVIEW_PRODUCT_CATEGORY
    , ENTITY
    , BU
    , SITE
    , WAREHOUSE_CODE
    , EXTENDED_COST
    , UNIT_COST
    , UOM
    , QUANTITY
    , UNIT_WEIGHT
    , WEIGHT_LBS
    , IS_ACQUIRED_DATE
    , IS_ACQUIRED_NOW
    , META_LOAD_DATETIME
    , RECORD_SOURCE
FROM `production`.`silver`.`bsat_combined_inventory_stock_levels`
WHERE DATE IS NOT NULL
) AS x
GROUP BY DATE
ORDER BY DATE DESC
;

SELECT DATE, COUNT(*)
FROM (
WITH sats AS (
    SELECT *
    FROM `production`.`silver`.`br_inventory_stock_levels_kinetic_step_1`

    UNION ALL

    SELECT *
    FROM `production`.`silver`.`br_combined_quickbooks_stock_status_step_1`
)

SELECT sats.*    
    , COALESCE(c.ACQUISITION_DATE, CAST('1900-01-01' AS DATE)) AS IS_ACQUIRED_DATE
    , CASE 
          WHEN COALESCE(c.ACQUISITION_DATE, CAST('1900-01-01' AS DATE)) <= sats.DATE THEN 1
          ELSE 0
      END AS IS_ACQUIRED_NOW
FROM sats
LEFT JOIN `production`.`silver`.`ref_master_ecore_companies` AS c
    ON UPPER(sats.ENTITY) = UPPER(c.COMPANY_CODE)
) AS x
GROUP BY DATE
ORDER BY DATE DESC
;
