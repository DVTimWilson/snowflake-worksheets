-- COUNT(*)  -- 92976
-- DISTINCT qb.companycode, qb.item  -- 841
-- DISTINCT qb.companycode, qb.item, mpc.item  -- 841
-- DISTINCT qb.companycode  -- 360, CRC, DGH, DOS, EDGE, ETI, ETR, HTI, PRC, PTS, TSS
-- DISTINCT mpc.COMPANY_ID  -- NULL, 360, CRC, DOS, ETI, ETR, PRC, TSS
      qb.QTY AS QUANTITY
     , CASE WHEN qb.COMPANYCODE = 'ETI' THEN UPPER(qb.UOM)
            ELSE CASE
  WHEN UPPER(TRIM(mpc.SOLD_BY_UNIT_OF_MEASURE)) IS NULL OR UPPER(TRIM(mpc.SOLD_BY_UNIT_OF_MEASURE)) IN ('', '-1', '0', 'NULL') THEN 'Uncategorized'
  ELSE COALESCE(
         element_at(
           map_from_arrays(
             array('EACH', 'LOAD'),
             array('EA', 'LD')
           ),
           UPPER(TRIM(mpc.SOLD_BY_UNIT_OF_MEASURE))
         ),
         UPPER(TRIM(mpc.SOLD_BY_UNIT_OF_MEASURE))
       )
END
       END AS UOM
     , CASE WHEN qb.COMPANYCODE = 'ETI' THEN CAST(qb.QTY AS DECIMAL(38,8))
            ELSE 

(
  
    CAST(COALESCE(TRY_MULTIPLY(qb.QTY, mpc.POUNDS_PER_SOLD_BY_UNIT) , 0) AS DECIMAL(38,8))
  
)
       END AS WEIGHT_LBS
,qb.*
,mpc.*
FROM combined_qb as qb

LEFT OUTER JOIN `tim_wilson`.`silver`.`ref_sat_master_product_categories_non_kinetic_latest` as mpc
    ON -- Case-insensitive equality for ad-hoc use
UPPER(qb.ITEM) = UPPER(mpc.ITEM)

LEFT OUTER JOIN (
    SELECT Business_Segment 
           , Company_Code
    FROM `tim_wilson`.`silver`.`ref_master_ecore_companies` AS x
) AS mec
    ON -- Case-insensitive equality for ad-hoc use
UPPER(qb.COMPANYCODE) = UPPER(mec.COMPANY_CODE)
    
LEFT OUTER JOIN `tim_wilson`.`silver`.`br_outbound_invoice_cogspc_pc_lookup` AS cogspc
ON DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.year
AND DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.quarter
AND UPPER(COALESCE(mpc.COREVIEW_PRODUCT_CATEGORY, 'Uncategorized')) = UPPER(cogspc.COREVIEW_PRODUCT_CATEGORY)

LEFT OUTER JOIN `tim_wilson`.`silver`.`br_outbound_invoice_cogspc_bu_lookup` AS cogsbu
ON DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogsbu.year
AND DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogsbu.quarter
AND -- Case-insensitive equality for ad-hoc use
UPPER(mec.BUSINESS_SEGMENT) = UPPER(cogsbu.BU)

WHERE 1=1
AND TYPE <> 'Inventory Adjust'
AND ACCOUNTTYPE <> 'Cost of Goods Sold'
AND mpc.COMPANY_ID IS NULL
