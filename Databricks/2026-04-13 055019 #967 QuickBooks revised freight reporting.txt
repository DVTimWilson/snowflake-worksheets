-- Back to QuickBooks
-- when NOT ETR AND invoiceline = 1 then AMOUNT aka SALES_DOLLAR + freight amount --> SALES_DOLLAR_WITH_FREIGHT
-- else AMOUNT aka SALES_DOLLAR --> SALES_DOLLAR_WITH_FREIGHT

WITH freight_data AS (
SELECT all.INVOICE_NUMBER, COUNT(all.INVOICELINE) AS all, COUNT(nonfreight.INVOICELINE) AS nonfreight, COUNT(freight.INVOICELINE) AS freight, SUM(all.SALES_DOLLAR) AS total_amount, COALESCE(SUM(nonfreight.SALES_DOLLAR), 0) AS non_freight_amount, COALESCE(SUM(freight.SALES_DOLLAR), 0) AS freight_amount
FROM tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS all

LEFT OUTER JOIN tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS nonfreight
ON all.INVOICE_NUMBER = nonfreight.INVOICE_NUMBER
AND all.INVOICELINE = nonfreight.INVOICELINE
AND nonfreight.ACCOUNT NOT ILIKE '%freight%'

LEFT OUTER JOIN tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS freight
ON all.INVOICE_NUMBER = freight.INVOICE_NUMBER
AND all.INVOICELINE = freight.INVOICELINE
AND freight.ACCOUNT ILIKE '%freight%'

GROUP BY all.INVOICE_NUMBER
-- HAVING nonfreight > 0
-- ORDER BY all.INVOICE_NUMBER
)

SELECT
CASE WHEN q.ENTITY != 'ETR' AND q.INVOICELINE = 1 THEN q.SALES_DOLLAR + COALESCE(f.freight_amount, 0)
ELSE SALES_DOLLAR 
END AS SALES_DOLLAR_WITH_FREIGHT
, q.SALES_DOLLAR
, f.freight_amount
, q.*
FROM tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS q

LEFT OUTER JOIN freight_data AS f
ON q.INVOICE_NUMBER = f.INVOICE_NUMBER

WHERE (q.ENTITY = 'ETR' OR q.ACCOUNT NOT ILIKE '%freight%')
AND f.freight_amount > 0

ORDER BY INVOICE_NUMBER, INVOICELINE
;

WITH non_freight_data AS (
SELECT q.INVOICE_NUMBER, q.ENTITY
, COUNT(*) AS non_freight_count
-- , SUM(q.SALES_DOLLAR) AS total_non_freight_sales_dollar
FROM tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS q
WHERE q.ACCOUNT NOT ILIKE '%freight%'
GROUP BY q.INVOICE_NUMBER, q.ENTITY
),

freight_data AS (
SELECT q.INVOICE_NUMBER, q.ENTITY
, COUNT(*) AS freight_count
, SUM(q.SALES_DOLLAR) AS total_freight_sales_dollar
FROM tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS q
WHERE q.ACCOUNT ILIKE '%freight%'
GROUP BY q.INVOICE_NUMBER, q.ENTITY
),

joined_data AS (
SELECT q.*
, COALESCE(n.non_freight_count, 0) AS non_freight_count
-- , n.total_non_freight_sales_dollar
, COALESCE(f.freight_count, 0) AS freight_count
, CASE WHEN q.ENTITY = 'ETR' OR q.INVOICELINE > 1 THEN 0
  ELSE COALESCE(f.total_freight_sales_dollar, 0)
  END AS total_freight_sales_dollar
FROM tim_wilson.silver.br_outbound_invoice_quickbooks_v2 AS q

LEFT OUTER JOIN non_freight_data AS n
ON q.INVOICE_NUMBER = n.INVOICE_NUMBER
AND q.ENTITY = n.ENTITY

LEFT OUTER JOIN freight_data AS f
ON q.INVOICE_NUMBER = f.INVOICE_NUMBER
AND q.ENTITY = f.ENTITY
)

SELECT *
, CASE WHEN ENTITY = 'ETR' THEN SALES_DOLLAR
  WHEN non_freight_count > 0 AND freight_count > 0 AND INVOICELINE = 1 THEN SALES_DOLLAR + total_freight_sales_dollar
  ELSE SALES_DOLLAR
  END AS sales_with_freight
FROM joined_data

WHERE 1 = 1
-- AND INVOICE_NUMBER = '30002'
-- AND ENTITY != 'ETR'
-- AND freight_count > 0
AND (ENTITY = 'ETR'
OR ACCOUNT NOT ILIKE '%freight%')

ORDER BY INVOICE_NUMBER, ENTITY, INVOICELINE
