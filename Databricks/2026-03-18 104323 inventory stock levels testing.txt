WITH old AS (
SELECT INVENTORY_ITEM_ID, DATE, SUM(QUANTITY) AS OLD_SUM_QUANTITY
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels
GROUP BY INVENTORY_ITEM_ID, DATE
),

new AS (
SELECT INVENTORY_ITEM_ID, DATE, SUM(QUANTITY) AS NEW_SUM_QUANTITY
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels_v2
GROUP BY INVENTORY_ITEM_ID, DATE
)

SELECT old.INVENTORY_ITEM_ID, old.DATE, old.OLD_SUM_QUANTITY, new.NEW_SUM_QUANTITY
FROM old
LEFT OUTER JOIN new
ON old.INVENTORY_ITEM_ID = new.inventory_item_id
AND old.DATE = new.date
WHERE old.OLD_SUM_QUANTITY != new.NEW_SUM_QUANTITY
ORDER BY old.INVENTORY_ITEM_ID, old.DATE
;

WITH old AS (
SELECT INVENTORY_ITEM_ID, DATE, SUM(QUANTITY) AS OLD_SUM_QUANTITY
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels
GROUP BY INVENTORY_ITEM_ID, DATE
),

new AS (
SELECT INVENTORY_ITEM_ID, DATE, SUM(QUANTITY) AS NEW_SUM_QUANTITY
FROM tim_wilson.gold_coreview.mart_inventory_stock_levels_v2
GROUP BY INVENTORY_ITEM_ID, DATE
)

SELECT SUM(old.OLD_SUM_QUANTITY), SUM(new.NEW_SUM_QUANTITY), SUM(new.NEW_SUM_QUANTITY) - SUM(old.OLD_SUM_QUANTITY)
FROM old
LEFT OUTER JOIN new
ON old.INVENTORY_ITEM_ID = new.inventory_item_id
AND old.DATE = new.date
;
