    JOIN (
        SELECT DISTINCT source_data.DEPARTMENT_HK
        FROM source_data
    ) AS source_records
        ON source_records.DEPARTMENT_HK = current_records.DEPARTMENT_HK
    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY current_records.DEPARTMENT_HK
        ORDER BY current_records.LOAD_DATETIME DESC
    ) = 1
),

unique_source_records AS (
    SELECT
        sd.DEPARTMENT_HK, sd.DEPARTMENT_HASHDIFF, sd.ALLOCATED_LEDGER_CODE, sd.DEPT, sd.ENTITY, sd.SITE, sd.LOAD_DATETIME, sd.RECORD_SOURCE
    FROM source_data AS sd
    LEFT OUTER JOIN latest_records AS lr
        ON sd.DEPARTMENT_HK = lr.DEPARTMENT_HK

    QUALIFY sd.DEPARTMENT_HASHDIFF !=
        LAG(sd.DEPARTMENT_HASHDIFF, 1,
            COALESCE(
                lr.HASHDIFF,
                CAST('FFFFFFFF' AS BINARY)
            )
        ) OVER (
            PARTITION BY sd.DEPARTMENT_HK
            ORDER BY sd.LOAD_DATETIME ASC
        )
),

WITH source_data AS (
  SELECT
    a.DEPARTMENT_HK,
    a.DEPARTMENT_HASHDIFF,
    a.ALLOCATED_LEDGER_CODE,
    a.DEPT,
    a.ENTITY,
    a.SITE,
    a.LOAD_DATETIME,
    a.RECORD_SOURCE
  FROM
    `tim_wilson`.`bronze`.`stg_department_dayforce` AS a
  WHERE
    a.DEPARTMENT_HK IS NOT NULL
),
latest_records AS (
  SELECT
    current_records.DEPARTMENT_HK,
    current_records.HASHDIFF,
    current_records.LOAD_DATETIME
  FROM
    `tim_wilson`.`silver`.`sat_department_dayforce` AS current_records
    LEFT SEMI JOIN source_data
        ON source_data.DEPARTMENT_HK = current_records.DEPARTMENT_HK
  QUALIFY
    ROW_NUMBER() OVER (
        PARTITION BY current_records.DEPARTMENT_HK
        ORDER BY current_records.LOAD_DATETIME DESC
      ) = 1
),
unique_source_records AS (
  SELECT
    sd.DEPARTMENT_HK,
    sd.DEPARTMENT_HASHDIFF,
    sd.ALLOCATED_LEDGER_CODE,
    sd.DEPT,
    sd.ENTITY,
    sd.SITE,
    sd.LOAD_DATETIME,
    sd.RECORD_SOURCE
  FROM
    source_data AS sd
      LEFT OUTER JOIN latest_records AS lr
        ON sd.DEPARTMENT_HK = lr.DEPARTMENT_HK
  QUALIFY
    sd.DEPARTMENT_HASHDIFF != LAG(
      sd.DEPARTMENT_HASHDIFF, 1, COALESCE(lr.HASHDIFF, CAST('FFFFFFFF' AS BINARY))
    ) OVER (PARTITION BY sd.DEPARTMENT_HK ORDER BY sd.LOAD_DATETIME ASC)
),
records_to_insert AS (
  SELECT
    usr.DEPARTMENT_HK,
    usr.DEPARTMENT_HASHDIFF AS HASHDIFF,
    usr.ALLOCATED_LEDGER_CODE,
    usr.DEPT,
    usr.ENTITY,
    usr.SITE,
    usr.LOAD_DATETIME,
    usr.RECORD_SOURCE
  FROM
    unique_source_records AS usr
)

  *
FROM
  records_to_insert
