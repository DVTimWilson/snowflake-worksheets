SELECT *
FROM tim_wilson.gold_coreview.mart_outbound_transactions
WHERE 1=1
-- AND MONTH(INVOICE_DATE) = 6
AND YEAR(INVOICE_DATE) = 2026
AND RECORD_SOURCE = 'KINETIC.TRIALBALANCEREC'
ORDER BY INVOICE_NUMBER
;

SELECT SUM(SALES_DOLLAR)
FROM tim_wilson.gold_coreview.mart_outbound_transactions
WHERE 1=1
AND MONTH(INVOICE_DATE) = 6
AND YEAR(INVOICE_DATE) = 2026
AND RECORD_SOURCE = 'KINETIC.TRIALBALANCEREC'
AND ENTITY = 'SAI'
;

SELECT ENTITY, SUM(SALES_DOLLAR)
FROM tim_wilson.gold_coreview.mart_outbound_transactions
WHERE 1=1
AND MONTH(INVOICE_DATE) = 6
AND YEAR(INVOICE_DATE) = 2026
AND RECORD_SOURCE = 'KINETIC.TRIALBALANCEREC'
AND ENTITY IN ('SAI', 'STI')
GROUP BY ENTITY
;
