WITH combined_qb AS(

SELECT * FROM `tim_wilson`.`silver`.`sat_outbound_invoice_quickbooks`
UNION
SELECT * FROM `tim_wilson`.`silver`.`sat_outbound_invoice_quickbooks_dgh`
UNION
SELECT * FROM `tim_wilson`.`silver`.`sat_outbound_invoice_quickbooks_edge`
UNION
SELECT * FROM `tim_wilson`.`silver`.`sat_outbound_invoice_quickbooks_hti`

)

SELECT
     INVOICE_LINE_LHK
     , INVOICELINE
     , DATE AS INVOICEDATE
     , qb.NAME AS CUSTOMER
     , qb.ITEM as PRODUCT_ID
     , COALESCE(mpc.COREVIEW_PRODUCT_CATEGORY, 'UNCATEGORIZED') AS COREVIEW_PRODUCT_CATEGORY
     , qb.CLASS as SITE
     , mec.BUSINESS_SEGMENT as BU
     , COMPANYCODE AS ENTITY
     , AMOUNT AS SALES_DOLLAR
     , AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc) AS COGS_DOLLAR
     , TRY_DIVIDE((AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc)), QTY) AS UNIT_COST
     , AMOUNT - (AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc)) AS GROSS_MARGIN_DOLLAR
     , TRY_DIVIDE((AMOUNT - (AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc))), AMOUNT) AS GROSS_MARGIN_PCT
     , QTY AS QUANTITY
     , CASE
         WHEN cogspc.average_cogspc IS NOT NULL THEN 'PC'
         WHEN cogsbu.average_cogspc IS NOT NULL THEN 'BU'
         ELSE NULL
       END AS COGSPC_SOURCE
     , LOAD_DATETIME
     , RECORD_SOURCE
     , HASHDIFF
FROM combined_qb as qb

LEFT OUTER JOIN (SELECT Item, CoreView_Product_Category
FROM `landing`.`reftables`.`master_product_category_non_kinetic`
QUALIFY row_number() OVER(PARTITION BY ITEM ORDER BY CoreView_Product_Category) = 1) as mpc
    ON qb.item = mpc.item

LEFT OUTER JOIN `landing`.`reftables`.`master_ecore_companies` AS mec
    ON qb.companycode= mec.company_code

LEFT OUTER JOIN `tim_wilson`.`silver`.`br_outbound_invoice_cogspc_pc_lookup` AS cogspc
ON DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.year
AND DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.quarter
AND mpc.COREVIEW_PRODUCT_CATEGORY = cogspc.COREVIEW_PRODUCT_CATEGORY

LEFT OUTER JOIN `tim_wilson`.`silver`.`br_outbound_invoice_cogspc_bu_lookup` AS cogsbu
ON DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.year
AND DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) = cogspc.quarter
AND mec.BUSINESS_SEGMENT = cogsbu.BU
;

     , COALESCE(mpc.COREVIEW_PRODUCT_CATEGORY, 'UNCATEGORIZED') AS COREVIEW_PRODUCT_CATEGORY
     , qb.CLASS as SITE
     , mec.BUSINESS_SEGMENT as BU
     ,cogspc.average_cogspc AS average_cogspcPC, cogsbu.average_cogspc AS average_cogspcBU
     ,cogsbu.BU
     ,DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE))
     ,DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE))
     ,(CASE WHEN DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) < 2023 THEN 2023 ELSE DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) END)
     ,(CASE WHEN DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) < 2023 THEN 1 ELSE DATE_PART('QUARTER', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) END)
     , COMPANYCODE AS ENTITY
     , AMOUNT AS SALES_DOLLAR
     , AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc) AS COGS_DOLLAR
     , TRY_DIVIDE((AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc)), QTY) AS UNIT_COST
     , AMOUNT - (AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc)) AS GROSS_MARGIN_DOLLAR
     , TRY_DIVIDE((AMOUNT - (AMOUNT * COALESCE(cogspc.average_cogspc, cogsbu.average_cogspc))), AMOUNT) AS GROSS_MARGIN_PCT
     , QTY AS QUANTITY
     , CASE
         WHEN cogspc.average_cogspc IS NOT NULL THEN 'PC'
         WHEN cogsbu.average_cogspc IS NOT NULL THEN 'BU'
         ELSE NULL
       END AS COGSPC_SOURCE
     , LOAD_DATETIME
     , RECORD_SOURCE
     , HASHDIFF
FROM combined_qb as qb

LEFT OUTER JOIN (SELECT Item, CoreView_Product_Category
FROM `landing`.`reftables`.`master_product_category_non_kinetic`
QUALIFY row_number() OVER(PARTITION BY ITEM ORDER BY CoreView_Product_Category) = 1) as mpc
    ON qb.item = mpc.item

LEFT OUTER JOIN `landing`.`reftables`.`master_ecore_companies` AS mec
    ON qb.companycode= mec.company_code

AND UPPER(mpc.COREVIEW_PRODUCT_CATEGORY) = UPPER(cogspc.COREVIEW_PRODUCT_CATEGORY)

LEFT OUTER JOIN `tim_wilson`.`silver`.`br_outbound_invoice_cogspc_bu_lookup` AS cogsbu
ON (CASE WHEN DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) < 2023 THEN 2023 ELSE DATE_PART('YEAR', CAST(DATE_ADD(MONTH, -3, CAST(qb.date AS TIMESTAMP)) AS DATE)) END) = cogsbu.year

AND UPPER(mec.BUSINESS_SEGMENT) = UPPER(cogsbu.BU)

         WHEN cogspc.average_cogspc IS NOT NULL THEN 'PC'
