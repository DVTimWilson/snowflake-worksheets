SELECT MONTH(Date) AS MONTH, 'v1' AS VERSION, SUM(AMOUNT) AS TOTAL_AMOUNT
FROM landing.quickbooks.outbound_sales_360
GROUP BY MONTH(Date)

UNION ALL

SELECT MONTH(Date) AS MONTH, 'v2' AS VERSION, SUM(AMOUNT) AS TOTAL_AMOUNT

-- ORDER BY MONTH(Date)

;

SELECT *
FROM
(
SELECT MONTH(Date) AS MONTH, 'v1' AS VERSION, COUNT(*) AS COUNT
FROM landing.quickbooks.outbound_sales_360
GROUP BY MONTH(Date)
-- ORDER BY MONTH(Date)

UNION ALL

SELECT MONTH(Date) AS MONTH, 'v2' AS VERSION, COUNT(*) AS COUNT
FROM landing.quickbooks.outbound_sales_360_v2
GROUP BY MONTH(Date)
) AS sourcedata
PIVOT (SUM(COUNT) FOR VERSION IN ('v1', 'v2'))
ORDER BY MONTH DESC
;

SELECT *
FROM
(
SELECT MONTH(Date) AS MONTH, 'v1' AS VERSION, SUM(AMOUNT) AS TOTAL_AMOUNT
FROM develop.silver.sat_outbound_invoice_quickbooks_360_latest
GROUP BY MONTH(Date)

UNION ALL

SELECT MONTH(Date) AS MONTH, 'v2' AS VERSION, SUM(AMOUNT) AS TOTAL_AMOUNT
FROM (SELECT *
FROM landing.quickbooks.outbound_sales_360_v2
QUALIFY ROW_NUMBER() OVER(PARTITION BY NUM, INDEX ORDER BY EXTRACT_DATE DESC) = 1)
GROUP BY MONTH(Date)
) AS sourcedata

PIVOT (SUM(TOTAL_AMOUNT) FOR VERSION IN ('v1', 'v2'))
ORDER BY MONTH DESC
;
