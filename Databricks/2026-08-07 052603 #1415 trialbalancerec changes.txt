SELECT DISTINCT COMPANY_CODE, GL_ACCT_DISP, JOURNAL_CODE
FROM uat.silver.sat_trialbalancerec_kinetic_latest
WHERE COMPANY_CODE IN ('SAI')
        -- AND GL_ACCT_DISP LIKE '30-4%'
        AND JOURNAL_CODE = 'GJ'
ORDER BY COMPANY_CODE, GL_ACCT_DISP, JOURNAL_CODE
;

SELECT DISTINCT COMPANY_CODE, GL_ACCT_DISP, JOURNAL_CODE
FROM uat.silver.sat_trialbalancerec_kinetic_latest
WHERE
(
(COMPANY_CODE = 'SAI' AND GL_ACCT_DISP LIKE '30-4%')
OR
(COMPANY_CODE = 'STI' AND GL_ACCT_DISP LIKE '60-4%')
)
AND JOURNAL_CODE = 'GJ'
ORDER BY COMPANY_CODE, GL_ACCT_DISP, JOURNAL_CODE
;

SELECT *
FROM tim_wilson.gold_coreview.mart_outbound_transactions

AND MONTH(INVOICE_DATE) = 6
AND YEAR(INVOICE_DATE) = 2026
AND RECORD_SOURCE = 'KINETIC.TRIALBALANCEREC'
