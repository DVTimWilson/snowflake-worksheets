-- Date windows do not overlap
-- SQL overlapping events
-- Using T-SQL to find events that overlap (or don't) in SQL Server
-- https://www.mssqltips.com/sqlservertip/5854/using-tsql-to-find-events-that-overlap-or-dont-in-sql-server/
SELECT *
FROM uat.gold_sow4.dim_customer_typeii AS c1

INNER JOIN uat.gold_sow4.dim_customer_typeii AS c2
ON c1.CUSTOMER_HK = c2.CUSTOMER_HK
-- Don't self join same records
AND c1.DIM_START_DATE != c2.DIM_START_DATE
AND c1.DIM_END_DATE != c2.DIM_END_DATE
-- Join records with overlapping time periods
AND c1.DIM_START_DATE < c2.DIM_END_DATE
AND c1.DIM_END_DATE > c2.DIM_START_DATE

-- WHERE c1.CUSTOMER_HK = UNBASE64('Af0ICG99I9DMeKkxo+1b1g==')
-- WHERE c1.CUSTOMER_HK = UNBASE64('Akurd7lGKjT/pziWkoAsWQ==')
;
