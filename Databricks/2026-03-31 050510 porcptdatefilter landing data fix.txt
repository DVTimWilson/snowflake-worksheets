SELECT * 
FROM (
SELECT l.*
, ROW_NUMBER() OVER (
        PARTITION BY 
              l.RCVDTL_PACKSLIP
            , l.RCVDTL_COMPANY
            , l.VENDOR_NAME
            , l.RCVHEAD_PLANT
            , l.CALCULATED_BU
            , l.RCVDTL_PARTNUM
        ORDER BY l.EXTRACT_DATE DESC
    ) AS rn
    , p.EXTRACT_DATE
FROM landing.kinetic.porcptdatefilter AS l
INNER JOIN psa.kinetic.psa_porcptdatefilter AS p
ON l.RCVDTL_PACKSLIP = p.RCVDTL_PACKSLIP
  AND l.RCVDTL_COMPANY = p.RCVDTL_COMPANY
  AND l.VENDOR_NAME = p.VENDOR_NAME
  AND l.RCVHEAD_PLANT = p.RCVHEAD_PLANT
  AND l.CALCULATED_BU = p.CALCULATED_BU
  AND l.RCVDTL_PARTNUM = p.RCVDTL_PARTNUM
WHERE TO_DATE(l.EXTRACT_DATE) = '2026-03-31'
) AS innerselect
WHERE rn != 1
ORDER BY RcvDtl_PONum, rn
;
