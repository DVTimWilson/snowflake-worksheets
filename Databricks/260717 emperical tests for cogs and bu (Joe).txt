SELECT a.ENTITY,a.INVOICE_DATE, a.INVOICE_NUMBER, a.CUSTOMER,a.PRODUCT_ID, a.UNIT_COST AS NEW_COST, b.UNIT_COST AS OLD_COST
FROM joe_barter.gold_coreview.mart_outbound_transactions as a
INNER JOIN uat.gold_coreview.mart_outbound_transactions as b
ON a.UNIQUE_KEY = b.UNIQUE_KEY
WHERE 1=1
-- AND a.ENTITY = 'EII'
AND a.GROSS_MARGIN_DOLLAR != 0
AND ROUND(a.UNIT_COST,2) != ROUND(b.UNIT_COST,2)
AND b.UNIT_COST <> 0
ORDER BY a.UNIQUE_KEY
;

SELECT DISTINCT kin.COMPANY, plant.CITY, kin.PARTNUM, stdcosts.STD_TOTAL_COST, kin.DOCUNITCOST
FROM joe_barter.silver.sat_outbound_invoice_kinetic_latest as kin
LEFT OUTER JOIN `joe_barter`.`silver`.`sat_plant_kinetic_latest` as plant
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(plant.COMPANY)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.PLANT) = UPPER(plant.PLANT)
INNER JOIN joe_barter.silver.ref_sat_bu_site_part_costs_latest as stdcosts 
    ON -- Case-insensitive equality for ad-hoc use
UPPER(kin.COMPANY) = UPPER(stdcosts.COMPANY)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.PLANT) = UPPER(stdcosts.SITE)
-- UPPER(COALESCE(NULLIF(plant.CITY, ''), 'Uncategorized')) = UPPER(stdcosts.SITE)
    AND -- Case-insensitive equality for ad-hoc use
UPPER(kin.PARTNUM) = UPPER(stdcosts.PART)
WHERE plant.PLANT IN ('GREENVL', 'MEXIA', 'BARBRTON')
AND kin.COMPANY IN ('ECS', 'ETC', 'ERP')

-- AND stdcosts.STD_TOTAL_COST <> 0
ORDER BY kin.PARTNUM
