-- develop.silver.dim_calendar_usa
-- Side-by-side comparison of record counts by PartTran_TranDate
WITH date_spine AS (
  SELECT date_day, day_name
  FROM develop.silver.dim_calendar_usa
  WHERE date_day BETWEEN '2026-07-24' AND '2026-09-11'
),
src_0725_0908 AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing_workato_test.kinetic.coparts_20260725_to_20260908
  GROUP BY CAST(PartTran_TranDate AS DATE)
),
src_0728_0911 AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing_workato_test.kinetic.coparts_20260728_to_20260911
  GROUP BY CAST(PartTran_TranDate AS DATE)
),
src_0801_0915 AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing_workato_test.kinetic.coparts_20260801_to_20260915
  GROUP BY CAST(PartTran_TranDate AS DATE)
),
src_new AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing_workato_test.kinetic.coparts_new
  GROUP BY CAST(PartTran_TranDate AS DATE)
),
src_landing AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing.kinetic.coparts
  GROUP BY CAST(PartTran_TranDate AS DATE)
),
src_landing_original AS (
  SELECT CAST(PartTran_TranDate AS DATE) AS tran_date, COUNT(*) AS cnt
  FROM landing.backup.coparts_20260015
  GROUP BY CAST(PartTran_TranDate AS DATE)
)
SELECT
  d.date_day,
  d.day_name,
  a.cnt AS coparts_0725_0908,
  b.cnt AS coparts_0728_0911,
  f.cnt AS coparts_0801_0915,
  c.cnt AS coparts_new,
  g.cnt AS landing_coparts_original,
  e.cnt AS landing_coparts
FROM date_spine d
LEFT OUTER JOIN src_0725_0908 a ON d.date_day = a.tran_date
LEFT OUTER JOIN src_0728_0911 b ON d.date_day = b.tran_date
LEFT OUTER JOIN src_0801_0915 f ON d.date_day = f.tran_date
LEFT OUTER JOIN src_new c ON d.date_day = c.tran_date
LEFT OUTER JOIN src_landing e ON d.date_day = e.tran_date
LEFT OUTER JOIN src_landing_original g ON d.date_day = g.tran_date
ORDER BY d.date_day
;
