WITH starting_balances AS (
  SELECT *
  FROM landing.kinetic.stockstat
  WHERE DATE(Extract_Date) = '2025-09-23'
),

movements AS (
  SELECT *
    , DATE(PartTran_TranDate) AS movement_date
    , PartTran_TranQty * -1 AS movement_qty_reversed
  FROM landing.raw_baq_explore.inventoryusage_test
),

calculations AS (
  SELECT *
    , SUM(movement_qty_reversed) OVER (PARTITION BY PartTran_PartNum ORDER BY movement_date ASC RANGE BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS total_movement_to_date
  FROM movements
)

SELECT *
  , movement_date
  , COALESCE(PartBin_OnhandQty, 0) AS starting_balance
  , total_movement_to_date
  , COALESCE(PartBin_OnhandQty, 0) + total_movement_to_date AS onhand_qty
FROM calculations AS c

LEFT OUTER JOIN starting_balances AS s
ON c.PartTran_PartNum = s.PartBin_PartNum
AND c.PartTran_Company = s.PartBin_Company
AND c.PartTran_WareHouseCode = s.PartBin_WarehouseCode
AND DATE(s.Extract_Date) = '2025-09-23'

ORDER BY movement_date, parttran_partnum
;
