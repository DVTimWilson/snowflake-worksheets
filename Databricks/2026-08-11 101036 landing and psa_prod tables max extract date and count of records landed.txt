BEGIN
  DECLARE v_date TIMESTAMP;
  DECLARE v_count BIGINT;
  DECLARE v_results ARRAY<STRUCT<table_schema: STRING, table_name: STRING, max_extract_date: TIMESTAMP, row_count_landed: BIGINT>> DEFAULT ARRAY();

  FOR t AS
    SELECT table_schema, table_name
    FROM landing.information_schema.tables
    WHERE table_schema NOT IN ('backup', 'information_schema')
      AND table_type NOT IN ('VIEW', 'MATERIALIZED_VIEW')
      AND LEFT(table_name, 2) != '__'
    --   AND LEFT(table_schema, 1) != '_'
      AND NOT table_name RLIKE '[0-9a-f]{8}_[0-9a-f]{4}_[0-9a-f]{4}_[0-9a-f]{4}_[0-9a-f]{12}'
      AND EXISTS (
        SELECT 1
        FROM landing.information_schema.columns c
        WHERE c.table_schema = tables.table_schema
          AND c.table_name = tables.table_name
          AND UPPER(c.column_name) = 'EXTRACT_DATE'
      )
    ORDER BY table_schema, table_name
  DO
    BEGIN
      DECLARE CONTINUE HANDLER FOR SQLEXCEPTION BEGIN SET v_date = TO_TIMESTAMP('1900-01-01 00:00:00.000000'); SET v_count = -1; END;
      EXECUTE IMMEDIATE
        CONCAT('SELECT MAX(EXTRACT_DATE), COUNT(*) FROM `landing`.`', t.table_schema, '`.`', t.table_name, '` WHERE EXTRACT_DATE = (SELECT MAX(EXTRACT_DATE) FROM `landing`.`', t.table_schema, '`.`', t.table_name, '`)')
        INTO v_date, v_count;
    END;
    SET v_results = ARRAY_APPEND(v_results, NAMED_STRUCT('table_schema', t.table_schema, 'table_name', t.table_name, 'max_extract_date', v_date, 'row_count_landed', v_count));
  END FOR;

  SELECT * FROM (SELECT inline(v_results)) AS x ORDER BY table_schema, table_name;
END;


SELECT EXTRACT_DATE, COUNT(*)
FROM landing.kinetic.partmaster
GROUP BY EXTRACT_DATE
ORDER BY EXTRACT_DATE DESC
;
