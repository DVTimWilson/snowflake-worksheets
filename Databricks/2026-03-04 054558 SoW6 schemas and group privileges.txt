-- Check privileges
WITH privileges AS (
  SELECT grantee, catalog_name, NULL AS schema_name, NULL AS table_name, privilege_type
  FROM system.information_schema.catalog_privileges

  UNION

  SELECT grantee, catalog_name, schema_name, NULL AS table_name, privilege_type
  FROM system.information_schema.schema_privileges

  UNION

  SELECT grantee, table_catalog as catalog_name, table_schema as schema_name, table_name, privilege_type
  FROM system.information_schema.table_privileges
)

SELECT * FROM privileges
-- WHERE grantee LIKE 'sow4%'
WHERE grantee LIKE 'sow6%'
-- WHERE grantee = 'coreview_user_prod'
ORDER BY grantee, catalog_name, schema_name, table_name, privilege_type
;
