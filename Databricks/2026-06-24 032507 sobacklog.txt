WITH uat AS (
SELECT ORDER_RELEASE_COMPANY, ORDER_HEADER_ORDER_DATE, COUNT(*) AS uatcount, SUM(calculated_sales) AS uatsum
FROM (SELECT * FROM uat.silver.sat_sobacklog_kinetic QUALIFY ROW_NUMBER() OVER (PARTITION BY UPPER(ORDER_RELEASE_COMPANY)
    , ORDER_HEADER_ORDER_NUMBER
    , ORDER_LINE_ORDER_LINE
    , ORDER_RELEASE_NUMBER
    ORDER BY EXTRACT_DATE DESC) = 1)
GROUP BY ORDER_RELEASE_COMPANY, ORDER_HEADER_ORDER_DATE
)
,

psa AS (
SELECT ORDER_RELEASE_COMPANY, ORDER_HEADER_ORDER_DATE, COUNT(*) AS psacount, SUM(calculated_sales) AS psasum
FROM (SELECT * FROM psa.kinetic.psa_kinetic_sobacklog QUALIFY ROW_NUMBER() OVER (PARTITION BY UPPER(ORDER_RELEASE_COMPANY)
    , ORDER_HEADER_ORDER_NUMBER
    , ORDER_LINE_ORDER_LINE
    , ORDER_RELEASE_NUMBER
    ORDER BY EXTRACT_DATE DESC) = 1)
GROUP BY ORDER_RELEASE_COMPANY, ORDER_HEADER_ORDER_DATE
)

SELECT uat.*, psa.psacount, psa.psasum, uat.uatsum - psa.psasum AS diff
FROM uat

FULL OUTER JOIN psa
ON uat.order_release_company = psa.order_release_company
AND uat.order_header_order_date = psa.order_header_order_date

ORDER BY uat.order_release_company ASC, uat.order_header_order_date DESC
;

WITH uat AS (
SELECT ORDER_RELEASE_COMPANY, EXTRACT_DATE AS uatextractdate, COUNT(*) AS uatcount, SUM(calculated_sales) AS uatsum
FROM uat.silver.sat_sobacklog_kinetic
GROUP BY ORDER_RELEASE_COMPANY, EXTRACT_DATE
)
,

psa AS (
SELECT ORDER_RELEASE_COMPANY, EXTRACT_DATE AS psaextractdate, COUNT(*) AS psacount, SUM(calculated_sales) AS psasum
FROM (SELECT * FROM psa.kinetic.psa_kinetic_sobacklog QUALIFY ROW_NUMBER() OVER (PARTITION BY UPPER(ORDER_RELEASE_COMPANY)
    , ORDER_HEADER_ORDER_NUMBER
    , ORDER_LINE_ORDER_LINE
    , ORDER_RELEASE_NUMBER
    ORDER BY EXTRACT_DATE DESC) = 1)
GROUP BY ORDER_RELEASE_COMPANY, EXTRACT_DATE
)

SELECT uat.*, psa.order_release_company, psa.psaextractdate, psa.psacount, psa.psasum, uat.uatsum - psa.psasum AS diff
FROM uat

FULL OUTER JOIN psa
ON uat.order_release_company = psa.order_release_company
AND uat.uatextractdate = psa.psaextractdate

ORDER BY uat.order_release_company ASC, uat.uatextractdate DESC
;
