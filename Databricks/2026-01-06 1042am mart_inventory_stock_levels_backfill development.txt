    , CASE WHEN DATE IS NULL THEN LAG(DATE) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE DATE
      END AS DATE

      ELSE INVENTORY_ITEM_DESCRIPTION
      END AS INVENTORY_ITEM_DESCRIPTION
    , CASE WHEN DATE IS NULL THEN LAG(COREVIEW_PRODUCT_CATEGORY) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE COREVIEW_PRODUCT_CATEGORY
      END AS COREVIEW_PRODUCT_CATEGORY
    , CASE WHEN DATE IS NULL THEN LAG(ENTITY) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE ENTITY
      END AS ENTITY
    , CASE WHEN DATE IS NULL THEN LAG(BU) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE BU
      END AS BU
    , CASE WHEN DATE IS NULL THEN LAG(SITE) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE SITE
      END AS SITE
    , CASE WHEN DATE IS NULL THEN LAG(TOTAL_VALUE) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE TOTAL_VALUE
      END AS TOTAL_VALUE
    , CASE WHEN DATE IS NULL THEN LAG(UNIT_COST) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE UNIT_COST
      END AS UNIT_COST
    , CASE WHEN DATE IS NULL THEN LAG(UOM) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE UOM
      END AS UOM
    , CASE WHEN DATE IS NULL THEN LAG(QUANTITY) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE QUANTITY
      END AS QUANTITY
    , CASE WHEN DATE IS NULL THEN LAG(UNIT_WEIGHT) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE UNIT_WEIGHT
      END AS UNIT_WEIGHT
    , CASE WHEN DATE IS NULL THEN LAG(WEIGHT_LBS) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE WEIGHT_LBS
      END AS WEIGHT_LBS
    , CASE WHEN DATE IS NULL THEN LAG(IS_ACQUIRED_DATE) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE IS_ACQUIRED_DATE
      END AS IS_ACQUIRED_DATE
    , CASE WHEN DATE IS NULL THEN LAG(IS_ACQUIRED_NOW) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE IS_ACQUIRED_NOW
      END AS IS_ACQUIRED_NOW
    , CASE WHEN META_LOAD_DATETIME IS NULL THEN LAG(META_LOAD_DATETIME) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE META_LOAD_DATETIME
      END AS META_LOAD_DATETIME
    , CASE WHEN DATE IS NULL THEN LAG(RECORD_SOURCE) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE RECORD_SOURCE
      END AS RECORD_SOURCE
    , CASE WHEN DATE IS NULL THEN LAG(HASHDIFF) IGNORE NULLS OVER (PARTITION BY dim_itemtype, dim_item, dim_entity ORDER BY dim_date)
      ELSE HASHDIFF
      END AS HASHDIFF
FROM rows_to_process

ORDER BY dim_date ASC, dim_itemtype, dim_item

 WITH inventory_with_gaps AS (
    SELECT 
        UNIQUE_KEY
        , DATE
        , INVENTORY_ITEM_ID
        , INVENTORY_ITEM_DESCRIPTION
        , COREVIEW_PRODUCT_CATEGORY
        , ENTITY
        , BU
        , SITE
        , TOTAL_VALUE
        , UNIT_COST
        , UOM
        , QUANTITY
        , UNIT_WEIGHT
        , WEIGHT_LBS
        , IS_ACQUIRED_DATE
        , IS_ACQUIRED_NOW
        , META_LOAD_DATETIME
        , RECORD_SOURCE
        , HASHDIFF
    FROM `tim_wilson`.`silver`.`bsat_combined_inventory_stock_levels`
    WHERE INVENTORY_ITEM_DESCRIPTION = '01 Inventory Parts (Inventory Part)'
--    WHERE INVENTORY_ITEM_DESCRIPTION = '01 Inventory Parts (Inventory Part)'  AND DATE <= '2025-02-11'
--    WHERE INVENTORY_ITEM_DESCRIPTION IN (SELECT DISTINCT INVENTORY_ITEM_DESCRIPTION FROM tim_wilson.silver.bsat_combined_inventory_stock_levels LIMIT 2)
--    WHERE INVENTORY_ITEM_ID = 'SAMP-005305' AND DATE <= '2025-09-25'
--    QUALIFY ROW_NUMBER () OVER (PARTITION BY DATE, INVENTORY_ITEM_ID, INVENTORY_ITEM_DESCRIPTION, ENTITY, SITE ORDER BY QUANTITY DESC) = 1
    ORDER BY DATE
)
-- SELECT COUNT(*) FROM inventory_with_gaps  -- 902591
-- SELECT * FROM inventory_with_gaps ORDER BY DATE, UNIQUE_KEY
,

distinct_inventory_items AS (
  SELECT DISTINCT
    INVENTORY_ITEM_ID
    , INVENTORY_ITEM_DESCRIPTION 
    , ENTITY
    , SITE
  FROM inventory_with_gaps
),

cross_join_data AS (
  -- DISTINCT ?
  -- DATE
  SELECT cal.date_day AS dim_date, INVENTORY_ITEM_ID AS dim_inventory_item_id, INVENTORY_ITEM_DESCRIPTION AS dim_inventory_item_description, ENTITY AS dim_entity, SITE AS dim_site
  FROM tim_wilson.silver.dim_calendar_usa AS cal

--  CROSS JOIN (SELECT DATE, INVENTORY_ITEM_ID, INVENTORY_ITEM_DESCRIPTION, ENTITY, SITE FROM inventory_with_gaps) AS inv
--  CROSS JOIN inventory_with_gaps AS inv
  CROSS JOIN distinct_inventory_items AS inv
)
-- SELECT * FROM cross_join_data WHERE dim_date >= '2025-02-11' ORDER BY dim_date, date
,

left_join_data AS (
  SELECT *
  FROM cross_join_data AS cjd

  LEFT OUTER JOIN inventory_with_gaps AS inv
  ON  1=1
  AND cjd.dim_date = inv.DATE
  AND cjd.dim_inventory_item_id = inv.INVENTORY_ITEM_ID
  AND cjd.dim_inventory_item_description = inv.INVENTORY_ITEM_DESCRIPTION
  AND cjd.dim_entity = inv.ENTITY
  AND cjd.dim_site = inv.SITE
)
--SELECT * FROM left_join_data WHERE dim_date >= '2025-02-11' ORDER BY dim_date, date
,

mart AS (
SELECT
    dim_date
    , date
    , dim_inventory_item_id
    , dim_inventory_item_description
    , dim_entity
    , dim_site
    , CASE WHEN DATE IS NULL THEN LAG(UNIQUE_KEY) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE UNIQUE_KEY
      END AS UNIQUE_KEY
    , CASE WHEN DATE IS NULL THEN LAG(DATE) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE DATE
      END AS DATE
    , CASE WHEN DATE IS NULL THEN LAG(INVENTORY_ITEM_ID) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE INVENTORY_ITEM_ID
      END AS INVENTORY_ITEM_ID
    , CASE WHEN DATE IS NULL THEN LAG(INVENTORY_ITEM_DESCRIPTION) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE INVENTORY_ITEM_DESCRIPTION
      END AS INVENTORY_ITEM_DESCRIPTION
    , CASE WHEN DATE IS NULL THEN LAG(COREVIEW_PRODUCT_CATEGORY) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE COREVIEW_PRODUCT_CATEGORY
      END AS COREVIEW_PRODUCT_CATEGORY
    , CASE WHEN DATE IS NULL THEN LAG(ENTITY) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE ENTITY
      END AS ENTITY
    , CASE WHEN DATE IS NULL THEN LAG(BU) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE BU
      END AS BU
    , CASE WHEN DATE IS NULL THEN LAG(SITE) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE SITE
      END AS SITE
    , CASE WHEN DATE IS NULL THEN LAG(TOTAL_VALUE) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE TOTAL_VALUE
      END AS TOTAL_VALUE
    , CASE WHEN DATE IS NULL THEN LAG(UNIT_COST) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE UNIT_COST
      END AS UNIT_COST
    , CASE WHEN DATE IS NULL THEN LAG(UOM) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE UOM
      END AS UOM
    , CASE WHEN DATE IS NULL THEN LAG(QUANTITY) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE QUANTITY
      END AS QUANTITY
    , CASE WHEN DATE IS NULL THEN LAG(UNIT_WEIGHT) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE UNIT_WEIGHT
      END AS UNIT_WEIGHT
    , CASE WHEN DATE IS NULL THEN LAG(WEIGHT_LBS) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE WEIGHT_LBS
      END AS WEIGHT_LBS
    , CASE WHEN DATE IS NULL THEN LAG(IS_ACQUIRED_DATE) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE IS_ACQUIRED_DATE
      END AS IS_ACQUIRED_DATE
    , CASE WHEN DATE IS NULL THEN LAG(IS_ACQUIRED_NOW) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE IS_ACQUIRED_NOW
      END AS IS_ACQUIRED_NOW
    , CASE WHEN META_LOAD_DATETIME IS NULL THEN LAG(META_LOAD_DATETIME) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE META_LOAD_DATETIME
      END AS META_LOAD_DATETIME
    , CASE WHEN DATE IS NULL THEN LAG(RECORD_SOURCE) IGNORE NULLS OVER (PARTITION BY dim_inventory_item_id, dim_inventory_item_description, dim_entity, dim_site ORDER BY dim_date)
      ELSE RECORD_SOURCE
      END AS RECORD_SOURCE
FROM left_join_data

WHERE dim_inventory_item_description = '01 Inventory Parts (Inventory Part)' AND dim_date >= '2025-02-11'
-- WHERE dim_date >= '2024-10-08'
-- WHERE dim_date >= '2025-09-23'
-- WHERE dim_date >= '2025-02-11'

ORDER BY dim_date, entity, site, inventory_item_id, inventory_item_description
)

WITH lowercase_data AS (
  SELECT a
  FROM VALUES ('a'), ('b'), ('c') AS ltestcol(a)
),

uppercase_data AS (
  SELECT b
  FROM VALUES ('A'), ('B'), ('C') AS utestcol(b)
)

SELECT a
FROM lowercase_data

UNION

SELECT b
FROM uppercase_data

ORDER BY 1
