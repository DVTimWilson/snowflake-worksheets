DESCRIBE TABLE tim_wilson.psa_kinetic.psa_porcptdatefilter
;



-- Define date range
WITH cte_dates AS (
  SELECT explode(sequence(to_date('2010-01-01'), to_date('2050-12-31'), interval 1 day)) AS dim_date
),

years AS (
  SELECT year(dim_date) AS cal_year
  FROM cte_dates
  GROUP BY cal_year
),

federal_holidays AS (
  SELECT
    make_date(cal_year, 1, 1) AS new_years_day,
    date_add(date_trunc('month', make_date(cal_year, 1, 1)), 
             ((7 + 1 - dayofweek(date_trunc('month', make_date(cal_year, 1, 1)))) % 7) + 14) AS birthday_of_martin_luther_king_jr,
    date_add(date_trunc('month', make_date(cal_year, 2, 1)), 
             ((7 + 1 - dayofweek(date_trunc('month', make_date(cal_year, 2, 1)))) % 7) + 14) AS washingtons_birthday,
    date_add(date_trunc('month', make_date(cal_year, 6, 1)), 
             ((7 + 1 - dayofweek(date_trunc('month', make_date(cal_year, 6, 1)))) % 7) - 7) AS memorial_day,
    CASE WHEN cal_year >= 2021 THEN make_date(cal_year, 6, 19) ELSE NULL END AS juneteenth,
    make_date(cal_year, 7, 4) AS independence_day,
    date_add(date_trunc('month', make_date(cal_year, 9, 1)), 
             ((7 + 1 - dayofweek(date_trunc('month', make_date(cal_year, 9, 1)))) % 7)) AS labor_day,
    date_add(date_trunc('month', make_date(cal_year, 10, 1)), 
             ((7 + 1 - dayofweek(date_trunc('month', make_date(cal_year, 10, 1)))) % 7) + 7) AS columbus_day,
    make_date(cal_year, 11, 11) AS veterans_day,
    date_add(date_trunc('month', make_date(cal_year, 11, 1)), 
             ((7 + 4 - dayofweek(date_trunc('month', make_date(cal_year, 11, 1)))) % 7) + 21) AS thanksgiving_day,
    make_date(cal_year, 12, 25) AS christmas_day
  FROM years
),

federal_holidays_observed AS (
  SELECT
    CASE WHEN date_format(new_years_day, 'E') = 'Sat' THEN date_add(new_years_day, -1)
         WHEN date_format(new_years_day, 'E') = 'Sun' THEN date_add(new_years_day, 1)
         ELSE new_years_day END AS new_years_day,
    birthday_of_martin_luther_king_jr,
    washingtons_birthday,
    memorial_day,
    CASE WHEN date_format(juneteenth, 'E') = 'Sat' THEN date_add(juneteenth, -1)
         WHEN date_format(juneteenth, 'E') = 'Sun' THEN date_add(juneteenth, 1)
         ELSE juneteenth END AS juneteenth,
    CASE WHEN date_format(independence_day, 'E') = 'Sat' THEN date_add(independence_day, -1)
         WHEN date_format(independence_day, 'E') = 'Sun' THEN date_add(independence_day, 1)
         ELSE independence_day END AS independence_day,
    labor_day,
    columbus_day,
    CASE WHEN date_format(veterans_day, 'E') = 'Sat' THEN date_add(veterans_day, -1)
         WHEN date_format(veterans_day, 'E') = 'Sun' THEN date_add(veterans_day, 1)
         ELSE veterans_day END AS veterans_day,
    thanksgiving_day,
    CASE WHEN date_format(christmas_day, 'E') = 'Sat' THEN date_add(christmas_day, -1)
         WHEN date_format(christmas_day, 'E') = 'Sun' THEN date_add(christmas_day, 1)
         ELSE christmas_day END AS christmas_day
  FROM federal_holidays
),

federal_holidays_unpivot_m AS (
  SELECT holiday_date, holiday_name
  FROM federal_holidays_observed
  LATERAL VIEW inline(array(
    struct(CAST(new_years_day AS DATE), 'new_years_day'),
    struct(CAST(birthday_of_martin_luther_king_jr AS DATE), 'birthday_of_martin_luther_king_jr'),
    struct(CAST(washingtons_birthday AS DATE), 'washingtons_birthday'),
    struct(CAST(memorial_day AS DATE), 'memorial_day'),
    struct(CAST(juneteenth AS DATE), 'juneteenth'),
    struct(CAST(independence_day AS DATE), 'independence_day'),
    struct(CAST(labor_day AS DATE), 'labor_day'),
    struct(CAST(columbus_day AS DATE), 'columbus_day'),
    struct(CAST(veterans_day AS DATE), 'veterans_day'),
    struct(CAST(thanksgiving_day AS DATE), 'thanksgiving_day'),
    struct(CAST(christmas_day AS DATE), 'christmas_day')
  )) holidays AS holiday_date, holiday_name
),

federal_holidays_unpivot AS (
    SELECT holiday_date, holiday_name
    FROM
    (
    SELECT
        new_years_day
        ,birthday_of_martin_luther_king_jr
        ,washingtons_birthday
        ,memorial_day
        ,juneteenth
        ,independence_day
        ,labor_day
        ,columbus_day
        ,veterans_day
        ,thanksgiving_day
        ,christmas_day
    FROM federal_holidays_observed
    ) AS unpivot_source
    UNPIVOT
    (
        holiday_date FOR holiday_name IN
            (
                new_years_day
                ,birthday_of_martin_luther_king_jr
                ,washingtons_birthday
                ,memorial_day
                ,juneteenth
                ,independence_day
                ,labor_day
                ,columbus_day
                ,veterans_day
                ,thanksgiving_day
                ,christmas_day
            )
    ) AS unpivot
),

calendar_dim AS (
  SELECT
    dim_date,
    date_trunc('week', dim_date) AS week_commencing_date,
    date_format(dim_date, 'yyyyMM') AS year_month,
    CASE WHEN date_format(dim_date, 'E') IN ('Sat', 'Sun') OR holiday_date IS NOT NULL THEN 0 ELSE 1 END AS working_day_flag,
    SUM(CASE WHEN date_format(dim_date, 'E') IN ('Sat', 'Sun') OR holiday_date IS NOT NULL THEN 0 ELSE 1 END)
      OVER (ORDER BY dim_date) AS working_day_index,
    holiday_name,
    date_format(dim_date, 'EEEE') AS day_name,
    date_format(dim_date, 'E') AS day_name_abv,
    dayofweek(dim_date) AS day_of_week,
    month(dim_date) AS month_num,
    concat(year(dim_date), date_format(dim_date, 'MM')) AS month_id,
    date_format(dim_date, 'MMMM') AS month_name,
    date_format(dim_date, 'MMM') AS month_name_abv,
    date_add(last_day(add_months(dim_date, -1)), 1) AS first_day_month,
    last_day(dim_date) AS last_day_month,
    quarter(dim_date) AS quarter_num,
    concat(year(dim_date), 'Q', quarter(dim_date)) AS quarter_id,
    date_add(last_day(add_months(dim_date, -3)), 1) AS first_day_quarter,
    last_day(add_months(dim_date, 3 - month(dim_date) % 3)) AS last_day_quarter,
    year(dim_date) AS dim_year
  FROM cte_dates
  LEFT JOIN federal_holidays_unpivot_m
  ON dim_date = holiday_date
)

SELECT *
FROM calendar_dim
WHERE dim_year = 2025
ORDER BY dim_date ASC
;
