SELECT *
FROM TIM_WILSON.TIM_WILSON.WATERLEVELS
ORDER BY STARTED_AT_TS DESC
;

UPDATE TIM_WILSON.TIM_WILSON.waterlevels
            SET WATERLEVEL_TS = TO_TIMESTAMP(EXTRACT_DATE),
                LOAD_DATETIME_COLUMN_NAME = 'EXTRACT_DATE'
            FROM (SELECT MAX(EXTRACT_DATE) AS EXTRACT_DATE
                  FROM TIM_WILSON.BRONZE.stg_company
            ) AS src
            WHERE TABLE_NAME = 'TIM_WILSON.BRONZE.stg_company'
            AND STARTED_AT_TS = TO_TIMESTAMP('2025-05-29 13:04:41.624018+00:00')
;
