SELECT RESOURCE_GROUP_HK
      ,HASHDIFF
      ,RESGRPDEPT_DESCRIPTION
      ,RESGRPDEPT_JCDEPT
      ,LOAD_DATETIME
      ,RECORD_SOURCE
  FROM TIM_WILSON.SILVER.sat_resource_group_resgrpeff
QUALIFY ROW_NUMBER() OVER (PARTITION BY RESOURCE_GROUP_HK
      ,HASHDIFF
      ,RESGRPDEPT_DESCRIPTION
      ,RESGRPDEPT_JCDEPT
      ,LOAD_DATETIME
      ,RECORD_SOURCE
      ORDER BY
      RESOURCE_GROUP_HK
      ,HASHDIFF
      ,RESGRPDEPT_DESCRIPTION
      ,RESGRPDEPT_JCDEPT
      ,LOAD_DATETIME
      ,RECORD_SOURCE) > 1
;

                      , RESGRPDEPT_DESCRIPTION

                      , EXTRACT_DATE
                     FROM TIM_WILSON.BRONZE.SRC_RESGRPEFF
                     WHERE RESGRPDEPT_DESCRIPTION = 'Ozark')

FROM columns
