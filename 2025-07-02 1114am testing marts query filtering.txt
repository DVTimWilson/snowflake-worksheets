-- 102
with aggregate AS (
SELECT a.DEPARTMENT_HK
       , month(b.PERFROMDATE) AS "MONTH"
       , year(b.PERFROMDATE) AS "YEAR"
       , sum(b.CREDIT1) AS CREDIT
       , sum(b.DEBIT1) AS DEBIT
       , sum(b.FISCALPERCB) AS COST

FROM TIM_WILSON.SILVER.lnk_trial_balance AS a

INNER JOIN TIM_WILSON.SILVER.ref_hub_natural AS d
ON d.NATURAL_HK = a.NATURAL_HK
AND UPPER(d.NATURAL_ID) LIKE '%LABOR%'

INNER JOIN TIM_WILSON.GOLD_MFG.br_latest_trial_balance AS b
ON a.TRIAL_BALANCE_LHK = b.TRIAL_BALANCE_LHK

GROUP BY a.department_hk, month(b.PERFROMDATE), year(b.PERFROMDATE)
    )

SELECT c.DEPARTMENT_NAME ,aggregate.*
FROM aggregate

LEFT OUTER JOIN TIM_WILSON.SILVER.hub_department AS c
ON c.department_hk = aggregate.department_hk
;
